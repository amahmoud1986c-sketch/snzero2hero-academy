### Background Script sketch (optional)

```javascript
// Run as admin in scoped app HR Leave — adjust user sys_ids first
(function () {
  var empUser = 'PASTE_EMPLOYEE_SYS_ID';
  var mgrUser = 'PASTE_MANAGER_SYS_ID';

  var emp = new GlideRecord('x_786686_hr_leave_employee_v2');
  emp.initialize();
  emp.u_user = empUser;
  emp.u_employee_id = 'E-1001';
  emp.u_department = 'People Ops';
  emp.u_manager = mgrUser;
  emp.u_join_date = '2024-01-15';
  emp.u_grade = 'G5';
  var empId = emp.insert();

  var bal = new GlideRecord('x_786686_hr_leave_bal_v2');
  bal.initialize();
  bal.u_employee_ref = empId;
  bal.u_year = 2026;
  bal.u_leave_type = 'annual';
  bal.u_entitled = 20;
  bal.u_used = 0;
  bal.u_remaining = 20;
  bal.insert();

  gs.info('Seeded employee=' + empId);
})();
```
