/**
 * Script Include: HRLXProLeaveUtilsV2
 * Scope: x_786686_hr_leave
 * Application: Custom · Scope: x_786686_hr_leave · sys_id: 83d5298c932b8f107520703efaba1072
 * Client callable: true  (extends AbstractAjaxProcessor)
 * Purpose: Leave calculations + helpers (business days, approver, balance, audit).
 *
 * Create in ServiceNow:
 *   Studio > Create Application File > Script Include
 *   Name: HRLXProLeaveUtilsV2
 *   Client callable: checked
 */
var HRLXProLeaveUtilsV2 = Class.create();
HRLXProLeaveUtilsV2.prototype = Object.extendsObject(global.AbstractAjaxProcessor, {

    /**
     * Count business days (Mon-Fri) inclusive between two date strings.
     * @param {string} startDate - 'yyyy-MM-dd'
     * @param {string} endDate   - 'yyyy-MM-dd'
     * @return {number}
     */
    calculateBusinessDays: function (startDate, endDate) {
        if (!startDate || !endDate) return 0;
        var start = new GlideDateTime(startDate + ' 00:00:00');
        var end = new GlideDateTime(endDate + ' 00:00:00');
        if (end.before(start)) return 0;

        var days = 0;
        var cursor = new GlideDateTime(start);
        while (!cursor.after(end)) {
            var dow = cursor.getDayOfWeekLocalTime(); // 1=Mon ... 7=Sun
            if (dow >= 1 && dow <= 5) days++;
            cursor.addDaysLocalTime(1);
        }
        return days;
    },

    /**
     * Resolve the approver (employee's manager) and set it on the record.
     * @param {GlideRecord} currentRecord - leave request record
     */
    setApproverFromEmployee: function (currentRecord) {
        var empId = currentRecord.getValue('u_employee_ref');
        if (!empId) return;
        var emp = new GlideRecord('x_786686_hr_leave_employee_v2');
        if (emp.get(empId) && emp.u_manager) {
            currentRecord.u_approver_user = emp.getValue('u_manager');
        }
    },

    /**
     * Write an audit row when the state changes.
     */
    auditStateChange: function (leaveSysId, oldState, newState, eventType) {
        var a = new GlideRecord('x_786686_hr_leave_audit_v2');
        a.initialize();
        a.u_leave_ref = leaveSysId;
        a.u_old_state = oldState || '';
        a.u_new_state = newState || '';
        a.u_event_type = eventType || 'state_change';
        a.u_actor = gs.getUserID();
        a.insert();
    },

    /**
     * Recalculate remaining balance for an employee/year.
     */
    recalculateBalance: function (employeeSysId, year) {
        var used = this._sumApprovedDays(employeeSysId, year);
        var bal = new GlideRecord('x_786686_hr_leave_bal_v2');
        bal.addQuery('u_employee_ref', employeeSysId);
        bal.addQuery('u_year', year);
        bal.query();
        if (bal.next()) {
            bal.u_used = used;
            bal.u_remaining = parseInt(bal.getValue('u_entitled') || 0, 10) - used;
            bal.update();
        }
    },

    _sumApprovedDays: function (employeeSysId, year) {
        var total = 0;
        var gr = new GlideAggregate('x_786686_hr_leave_req_v2');
        gr.addQuery('u_employee_ref', employeeSysId);
        gr.addQuery('u_state', 'approved');
        gr.addQuery('u_start_date', '>=', year + '-01-01');
        gr.addQuery('u_start_date', '<=', year + '-12-31');
        gr.addAggregate('SUM', 'u_days_count');
        gr.query();
        if (gr.next()) total = parseInt(gr.getAggregate('SUM', 'u_days_count') || 0, 10);
        return total;
    },

    /**
     * GlideAjax entry point for client-side day calculation.
     */
    ajaxGetBusinessDays: function () {
        return this.calculateBusinessDays(
            this.getParameter('sysparm_start'),
            this.getParameter('sysparm_end')
        );
    },

    type: 'HRLXProLeaveUtilsV2'
});

/**
 * Script Include: HRLXProSecurityUtilsV2
 * Scope: x_786686_hr_leave
 * Application: Custom · Scope: x_786686_hr_leave · sys_id: 83d5298c932b8f107520703efaba1072
 * Client callable: false
 * Purpose: Centralized security checks used by ACLs.
 *
 * Used by ACL scripts, e.g.:
 *   answer = new x_786686_hr_leave.HRLXProSecurityUtilsV2().canReadLeave(current);
 */
var HRLXProSecurityUtilsV2 = Class.create();
HRLXProSecurityUtilsV2.prototype = {

    initialize: function () {
        this.userId = gs.getUserID();
    },

    /** HR admin can do anything. */
    isHrAdmin: function () {
        return gs.hasRole('x_786686_hr_leave.hr_admin');
    },

    /** Can the current user READ this leave request? */
    canReadLeave: function (current) {
        if (this.isHrAdmin()) return true;
        // owner
        if (current.getValue('u_requested_for_user') === this.userId) return true;
        // approver / manager
        if (current.getValue('u_approver_user') === this.userId) return true;
        // manager of the employee
        return this._isManagerOfEmployee(current.getValue('u_employee_ref'));
    },

    /** Can the current user WRITE this leave request? */
    canWriteLeave: function (current) {
        if (this.isHrAdmin()) return true;
        var state = current.getValue('u_state');
        // owner can edit only while draft/submitted
        if (current.getValue('u_requested_for_user') === this.userId) {
            return (state === 'draft' || state === 'submitted');
        }
        // approver can act when submitted
        if (current.getValue('u_approver_user') === this.userId) {
            return (state === 'submitted');
        }
        return false;
    },

    _isManagerOfEmployee: function (employeeSysId) {
        if (!employeeSysId) return false;
        var emp = new GlideRecord('x_786686_hr_leave_employee_v2');
        if (emp.get(employeeSysId)) {
            return emp.getValue('u_manager') === this.userId;
        }
        return false;
    },

    type: 'HRLXProSecurityUtilsV2'
};

 * Application: Custom · Scope: x_786686_hr_leave · sys_id: 83d5298c932b8f107520703efaba1072
 * Target table: x_786686_hr_leave_req_v2
 *
 * Maps producer variables to record fields. Runs server-side when submitted.
 * Variables expected: leave_type, start_date, end_date, reason
 */
(function () {

    current.u_requested_for_user = gs.getUserID();
    current.u_leave_type = producer.leave_type;
    current.u_start_date = producer.start_date;
    current.u_end_date = producer.end_date;
    current.u_reason = producer.reason;
    current.u_state = 'submitted';

    // Link to employee profile of the current user (if exists)
    var emp = new GlideRecord('x_786686_hr_leave_employee_v2');
    emp.addQuery('u_user', gs.getUserID());
    emp.query();
    if (emp.next()) {
        current.u_employee_ref = emp.getUniqueValue();
    }

    // BR1 will calculate days and set the approver on insert.

})();

/**
 * Application: Custom · Scope: x_786686_hr_leave · sys_id: 83d5298c932b8f107520703efaba1072
 * Scheduled Job: Annual Leave Accrual
 * Run: yearly (e.g. Jan 1)
 *
 * Creates a fresh balance row per employee for the new year if missing.
 * Default entitlement is read from a property (fallback 21 annual days).
 */
(function annualAccrual() {

    var year = new GlideDateTime().getYearLocalTime();
    var annualEntitlement = parseInt(gs.getProperty('x_786686_hr_leave.config.annual_days', '21'), 10);

    var emp = new GlideRecord('x_786686_hr_leave_employee_v2');
    emp.query();

    var created = 0;
    while (emp.next()) {
        // skip if a balance for this year/type already exists
        var existing = new GlideRecord('x_786686_hr_leave_bal_v2');
        existing.addQuery('u_employee_ref', emp.getUniqueValue());
        existing.addQuery('u_year', year);
        existing.addQuery('u_leave_type', 'annual');
        existing.query();
        if (existing.hasNext()) continue;

        var bal = new GlideRecord('x_786686_hr_leave_bal_v2');
        bal.initialize();
        bal.u_employee_ref = emp.getUniqueValue();
        bal.u_year = year;
        bal.u_leave_type = 'annual';
        bal.u_entitled = annualEntitlement;
        bal.u_used = 0;
        bal.u_remaining = annualEntitlement;
        bal.insert();
        created++;
    }

    gs.info('[HRLX] Annual accrual ' + year + ' created ' + created + ' balance rows.');
})();

/**
 * Script Include: HRLXProIntegrationUtilsV2
 * Scope: x_786686_hr_leave
 * Application: Custom · Scope: x_786686_hr_leave · sys_id: 83d5298c932b8f107520703efaba1072
 * Client callable: false
 * Purpose: Inbound create + Outbound push for leave integration.
 */
var HRLXProIntegrationUtilsV2 = Class.create();
HRLXProIntegrationUtilsV2.prototype = {

    initialize: function () {},

    /**
     * Inbound: create a leave request from a JSON payload (string).
     * @return {string} sys_id of created record
     */
    inboundCreateLeave: function (jsonString) {
        var data = {};
        try { data = JSON.parse(jsonString); } catch (e) { return ''; }

        var gr = new GlideRecord('x_786686_hr_leave_req_v2');
        gr.initialize();
        gr.u_employee_ref = this._findEmployeeByExternalId(data.employee_id);
        gr.u_requested_for_user = this._findUserByEmployee(data.employee_id);
        gr.u_leave_type = data.leave_type || 'annual';
        gr.u_start_date = data.start_date || '';
        gr.u_end_date = data.end_date || '';
        gr.u_reason = data.reason || '';
        gr.u_state = 'submitted';
        return gr.insert();
    },

    /**
     * Outbound: push an approved leave decision to ERP/Payroll.
     * Endpoint + auth come from system properties (no secrets in code).
     */
    outboundPushDecision: function (leaveSysId) {
        var gr = new GlideRecord('x_786686_hr_leave_req_v2');
        if (!gr.get(leaveSysId)) return false;

        var endpoint = gs.getProperty('x_786686_hr_leave.config.erp_endpoint');
        if (!endpoint) {
            gs.error('[HRLX] Missing property x_786686_hr_leave.config.erp_endpoint');
            return false;
        }

        try {
            var r = new sn_ws.RESTMessageV2();
            r.setEndpoint(endpoint);
            r.setHttpMethod('POST');
            r.setRequestHeader('Content-Type', 'application/json');
            r.setRequestBody(JSON.stringify({
                number: gr.getValue('u_number'),
                employee: gr.getDisplayValue('u_employee_ref'),
                type: gr.getValue('u_leave_type'),
                days: gr.getValue('u_days_count'),
                state: gr.getValue('u_state')
            }));
            var resp = r.execute();
            var code = resp.getStatusCode();
            gs.info('[HRLX] Outbound status ' + code);
            return (code >= 200 && code < 300);
        } catch (ex) {
            gs.error('[HRLX] Outbound error: ' + ex.message);
            return false;
        }
    },

    _findEmployeeByExternalId: function (employeeId) {
        var e = new GlideRecord('x_786686_hr_leave_employee_v2');
        e.addQuery('u_employee_id', employeeId);
        e.query();
        return e.next() ? e.getUniqueValue() : '';
    },

    _findUserByEmployee: function (employeeId) {
        var e = new GlideRecord('x_786686_hr_leave_employee_v2');
        e.addQuery('u_employee_id', employeeId);
        e.query();
        return e.next() ? e.getValue('u_user') : '';
    },

    type: 'HRLXProIntegrationUtilsV2'
};

/**
 * Application: Custom · Scope: x_786686_hr_leave · sys_id: 83d5298c932b8f107520703efaba1072
 * Scripted REST API Resource: Create Leave (Inbound)
 * API ID:        hrlx_api_v2
 * Resource:      POST /leave
 * Full path:     /api/x_786686_hr_leave/hrlx_api_v2/leave
 *
 * TEST FIRST (no Postman until this works):
 *   All > System Web Services > REST API Explorer  ($restapi.do)
 *   Namespace: x_786686_hr_leave | API: hrlx_api_v2 | Resource: leave | POST

 *   Explorer → namespace now → Table API → x_786686_hr_leave_req_v2 → POST
 *   See: src/server/rest/rest_api_explorer.md
 *
 * Create API record:
 *   All > System Web Services > Scripted REST APIs > New + Resource POST /leave
 */
(function process(request, response) {

    var data = request.body && request.body.data ? request.body.data : {};

    // Validate
    if (!data.employee_id) {
        response.setStatus(400);
        return response.setBody({ error: 'employee_id is required' });
    }
    if (!data.start_date || !data.end_date) {
        response.setStatus(400);
        return response.setBody({ error: 'start_date and end_date are required' });

    // Create via integration utility

    if (!sysId) {
        response.setStatus(422);
        return response.setBody({ error: 'Could not create leave request (employee not found?)' });
    }

    response.setStatus(201);
    response.setBody({ sys_id: sysId, message: 'Leave request created' });

})(request, response);

var HRLXAIAgentTools = Class.create();
HRLXAIAgentTools.prototype = {
    initialize: function() {
        this.SCOPE = 'x_786686_hr_leave';
        this.T_EMP = 'x_786686_hr_leave_employee_v2';
        this.T_REQ = 'x_786686_hr_leave_req_v2';
        this.T_BAL = 'x_786686_hr_leave_bal_v2';
        this.T_AUD = 'x_786686_hr_leave_audit_v2';
    },

    _getEmployeeByUser: function(userId) {
        var gr = new GlideRecord(this.T_EMP);
        gr.addQuery('user', userId);
        gr.setLimit(1);
        gr.query();
        return gr.next() ? gr : null;
    },

    getBalance: function(userId) {
        var emp = this._getEmployeeByUser(userId);
        if (!emp) return { error: 'employee_not_found' };
        var out = { annual: 0, sick: 0, parental: 0, unpaid: 0 };
        var bal = new GlideRecord(this.T_BAL);
        bal.addQuery('u_employee_ref', emp.getUniqueValue());
        bal.query();
        while (bal.next()) {
            var t = String(bal.u_leave_type);
            if (out.hasOwnProperty(t)) out[t] = Number(bal.u_remaining_days) || 0;
        }
        return out;
    },

    requestLeave: function(payload) {
        try {
            if (gs.getProperty(this.SCOPE + '.ai.enabled', 'true') !== 'true')
                return { ok: false, error: 'ai_disabled' };

            var emp = this._getEmployeeByUser(payload.user_id);
            if (!emp) return { ok: false, error: 'employee_not_found' };

            var start = new GlideDateTime(payload.start_date);
            var end = new GlideDateTime(payload.end_date);
            if (start.compareTo(end) > 0)
                return { ok: false, error: 'start_after_end' };

            var bal = this.getBalance(payload.user_id);
            var days = Math.ceil((end.getNumericValue() - start.getNumericValue()) / 86400000) + 1;
            if ((bal[payload.type] || 0) < days)
                return { ok: false, error: 'insufficient_balance', needed: days, available: bal[payload.type] || 0 };

            var req = new GlideRecord(this.T_REQ);
            req.initialize();
            req.u_requested_for_user = payload.user_id;
            req.u_employee_ref = emp.getUniqueValue();
            req.u_leave_type = payload.type;
            req.u_start_date = payload.start_date;
            req.u_end_date = payload.end_date;
            req.u_reason = payload.reason || '';
            req.u_state = 'submitted';
            req.u_source = 'ai_agent';
            var sysId = req.insert();
            this._audit('ai_request_leave', sysId, payload);
            return { ok: true, request_id: sysId, number: req.getValue('u_number') };
        } catch (e) {
            gs.error('HRLXAIAgentTools.requestLeave: ' + e);
            return { ok: false, error: 'exception' };
        }
    },

    checkPolicy: function(question) {
        var profile = gs.getProperty(this.SCOPE + '.ai_search.profile', 'hrlx_kb_profile');
        try {
            var sn = new sn_ais.AISearch();
            var results = sn.search(profile, question, { limit: 5, genai: true });
            return {
                answer: results.genai_answer || '',
                citations: results.citations || []
            };
        } catch (e) {
            return { answer: '', citations: [], error: String(e) };
        }
    },

    escalate: function(userId, summary) {
        var emp = this._getEmployeeByUser(userId);
        if (!emp) return { ok: false, error: 'employee_not_found' };
        var hr = new GlideRecord('sn_hr_core_case');
        if (!hr.isValid()) {
            // Fallback: create audit + notify
            this._audit('ai_escalate_no_hrsd', userId, { summary: summary });
            return { ok: true, case_id: null, note: 'HRSD not installed — audit only' };
        }
        hr.initialize();
        hr.opened_for = userId;
        hr.short_description = 'AI escalation: ' + (summary || '').substring(0, 80);
        hr.description = summary;
        hr.subject_person = userId;
        var sysId = hr.insert();
        this._audit('ai_escalate', sysId, { summary: summary });
        return { ok: true, case_id: sysId };
    },

    getMyRequests: function(userId, days) {
        var emp = this._getEmployeeByUser(userId);
        if (!emp) return [];
        var since = new GlideDateTime();
        since.addDaysLocalTime(-1 * (days || 90));
        var gr = new GlideRecord(this.T_REQ);
        gr.addQuery('u_employee_ref', emp.getUniqueValue());
        gr.addQuery('sys_created_on', '>=', since);
        gr.orderByDesc('sys_created_on');
        gr.query();
        var out = [];
        while (gr.next()) {
            out.push({
                id: gr.getUniqueValue(),
                number: String(gr.u_number),
                type: String(gr.u_leave_type),
                start: String(gr.u_start_date),
                end: String(gr.u_end_date),
                state: String(gr.u_state)
            });
        }
        return out;
    },

    cancelRequest: function(userId, requestId) {
        var gr = new GlideRecord(this.T_REQ);
        if (!gr.get(requestId)) return { ok: false, error: 'not_found' };
        if (gr.u_requested_for_user != userId) return { ok: false, error: 'not_owner' };
        if (gr.u_state != 'submitted') return { ok: false, error: 'not_cancellable' };
        gr.u_state = 'cancelled';
        gr.update();
        this._audit('ai_cancel_request', requestId, { user: userId });
        return { ok: true };
    },

    _audit: function(action, recordId, payload) {
        var a = new GlideRecord(this.T_AUD);
        a.initialize();
        a.u_action = action;
        a.u_record = recordId;
        a.u_payload = JSON.stringify(payload || {});
        a.u_actor = gs.getUserID();
        a.u_source = 'ai_agent';
        a.insert();
    },

    type: 'HRLXAIAgentTools'
};

var HRLXAIProvider = Class.create();
HRLXAIProvider.prototype = {
    initialize: function() {
        this.SCOPE = 'x_786686_hr_leave';
        this.TIMEOUT_MS = parseInt(gs.getProperty(this.SCOPE + '.llm.timeout_ms', '15000'), 10);
    },

    _prop: function(name, def) { return gs.getProperty(this.SCOPE + '.' + name, def); },

    _budgetCheck: function(provider) {
        var budget = parseFloat(this._prop('llm.daily_budget_usd', '10'));
        var key = 'llm.spent.' + provider + '.' + new GlideDateTime().getDate().getDisplayValue();
        return parseFloat(gs.getProperty(this.SCOPE + '.' + key, '0')) < budget;
    },

    askClaude: function(prompt, model, maxTokens, system) {
        if (!this._budgetCheck('anthropic')) return { error: 'budget_exceeded' };
        try {
            var rm = new sn_ws.RESTMessageV2('Claude_API', 'post');
            rm.setHttpTimeout(this.TIMEOUT_MS);
            rm.setStringParameterNoEscape('model', model || this._prop('llm.anthropic.model', 'claude-sonnet-4-5'));
            rm.setStringParameterNoEscape('max_tokens', String(maxTokens || 400));
            rm.setStringParameterNoEscape('system', system || 'You are a helpful HR assistant.');
            rm.setStringParameterNoEscape('user_message', prompt);
            var resp = rm.execute();
            if (resp.getStatusCode() >= 400) return { error: 'http_' + resp.getStatusCode() };
            var json = JSON.parse(resp.getBody());
            var text = '';
            (json.content || []).forEach(function(c) { if (c.type === 'text') text += c.text; });
            return { text: text, usage: json.usage || {} };
        } catch (e) { return { error: 'exception', message: String(e) }; }
    },

    askWithFallback: function(prompt, options) {
        options = options || {};
        var chain = options.chain || ['anthropic', 'openai', 'now_llm'];
        for (var i = 0; i < chain.length; i++) {
            var res = (chain[i] === 'anthropic') ? this.askClaude(prompt, options.model, options.max_tokens, options.system) : null;
            if (res && !res.error && res.text) return res;
        }
        return { error: 'all_providers_failed' };
    },

    type: 'HRLXAIProvider'
};

var HRLXAICache = Class.create();
HRLXAICache.prototype = {
    initialize: function() { this.TABLE = 'x_786686_hr_leave_ai_cache'; },
    _hash: function(s) { return new GlideDigest().getSHA256Base64(s); },
    get: function(skill, input) {
        var gr = new GlideRecord(this.TABLE);
        gr.addQuery('u_hash', this._hash(skill + '|' + input));
        gr.addQuery('u_expires_at', '>', new GlideDateTime());
        gr.setLimit(1);
        gr.query();
        return gr.next() ? String(gr.u_response) : null;
    },
    put: function(skill, input, response, ttlMin) {
        var gr = new GlideRecord(this.TABLE);
        gr.initialize();
        gr.u_hash = this._hash(skill + '|' + input);
        gr.u_skill = skill;
        gr.u_response = response;
        var exp = new GlideDateTime();
        exp.addSeconds(60 * (ttlMin || 60));
        gr.u_expires_at = exp;
        gr.insert();
    },
    type: 'HRLXAICache'
};
