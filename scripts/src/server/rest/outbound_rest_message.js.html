 * Production call options:

 *   B) RESTMessageV2('HRLX ERP Push Decision', 'Push Decision') + setStringParameter('employee_id', ...)

// --- Reference: call configured REST Message by name ---
function pushDecisionViaRestMessage(leaveSysId) {
    var gr = new GlideRecord('x_786686_hr_leave_req_v2');
    if (!gr.get(leaveSysId)) return;

    var empId = '';
    if (gr.u_employee_ref) {
        var e = gr.u_employee_ref.getRefRecord();
        if (e.isValidRecord()) empId = e.getValue('u_employee_id');
    }

    var rm = new sn_ws.RESTMessageV2('HRLX ERP Push Decision', 'Push Decision');
    rm.setStringParameter('employee_id', empId);
    var resp = rm.execute();
    gs.info('[HRLX] ERP push status: ' + resp.getStatusCode());
}

// --- Reference: raw outbound (no REST Message record) ---
function pushDecisionRaw(leaveSysId) {
    var gr = new GlideRecord('x_786686_hr_leave_req_v2');
    if (!gr.get(leaveSysId)) return;

    var r = new sn_ws.RESTMessageV2();
    r.setEndpoint(gs.getProperty('x_786686_hr_leave.config.erp_endpoint'));
    r.setHttpMethod('POST');
    r.setRequestHeader('Content-Type', 'application/json');
    r.setRequestBody(JSON.stringify({
        leave_number: gr.getValue('u_number'),
        state: gr.getValue('u_state'),
        start_date: gr.getValue('u_start_date'),
        end_date: gr.getValue('u_end_date'),
        decision_time: gr.getValue('sys_updated_on')
    }));
    var resp = r.execute();
    gs.info('[HRLX] ERP push status: ' + resp.getStatusCode());
}
