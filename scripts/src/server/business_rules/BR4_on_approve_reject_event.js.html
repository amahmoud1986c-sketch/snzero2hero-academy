/**
 * Business Rule: BR4 - Fire Approved / Rejected Events
 * Application: Custom · Scope: x_786686_hr_leave · sys_id: c84db0e8839f0b509f02b0d6feaad3c9
 * Table:  x_786686_hr_leave_req_v2
 * When:   after
 * Update: true
 * Insert: false
 * Order:  250
 *
 * Drives notifications for approved and rejected leave requests.
 * Requires events registered in Event Registry (see events.md).
 */
(function executeRule(current, previous) {

    if (current.u_state.changesTo('approved')) {
        gs.eventQueue(
            'x_786686_hr_leave.leave_v2.approved',
            current,
            current.getValue('u_requested_for_user'),
            current.getValue('u_number')
        );
    }

    if (current.u_state.changesTo('rejected')) {
        gs.eventQueue(
            'x_786686_hr_leave.leave_v2.rejected',
            current,
            current.getValue('u_requested_for_user'),
            current.getValue('u_number')
        );
    }

})(current, previous);
