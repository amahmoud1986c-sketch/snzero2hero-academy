(function executeRule(current, previous) {
    if (!gs.getProperty('x_786686_hr_leave.ai.enabled', 'true') === 'true') return;
    if (current.u_leave_type) return;
    if (!current.u_reason) return;

    try {
        var ext = new sn_one_extend.OneExtendUtil();
        var res = ext.execute({
            executionRequests: [{
                capabilityId: 'hrlx_classify_leave_type',
                payload: {
                    u_leave_type: String(current.u_leave_type),
                    u_reason: String(current.u_reason),
                    u_notes: String(current.u_notes || '')
                }
            }]
        });
        var label = (res.capabilities[0] || {}).response;
        if (/^(annual|sick|parental|unpaid)$/.test(label))
            current.u_leave_type = label;
    } catch (e) {
        gs.warn('hrlx_classify_leave_type failed: ' + e);
    }
})(current, previous);

        bal.query();
        while (bal.next()) {
            var t = String(bal.u_leave_type);
            if (out.hasOwnProperty(t)) out[t] = Number(bal.u_remaining_days) || 0;
        }
        return out;
    },

    requestLeave: function(payload) {
        try {
            if (gs.getProperty(this.SCOPE + '.ai.enabled', 'true') !== 'true')
                return { ok: false, error: 'ai_disabled' };

            var emp = this._getEmployeeByUser(payload.user_id);
            if (!emp) return { ok: false, error: 'employee_not_found' };

            var start = new GlideDateTime(payload.start_date);
            var end = new GlideDateTime(payload.end_date);
            if (start.compareTo(end) > 0)
                return { ok: false, error: 'start_after_end' };

            var bal = this.getBalance(payload.user_id);
            var days = Math.ceil((end.getNumericValue() - start.getNumericValue()) / 86400000) + 1;
            if ((bal[payload.type] || 0) < days)
                return { ok: false, error: 'insufficient_balance', needed: days, available: bal[payload.type] || 0 };

            var req = new GlideRecord(this.T_REQ);
            req.initialize();
            req.u_requested_for_user = payload.user_id;
            req.u_employee_ref = emp.getUniqueValue();
            req.u_leave_type = payload.type;
            req.u_start_date = payload.start_date;
            req.u_end_date = payload.end_date;
            req.u_reason = payload.reason || '';
            req.u_state = 'submitted';
            req.u_source = 'ai_agent';
            var sysId = req.insert();
            this._audit('ai_request_leave', sysId, payload);
            return { ok: true, request_id: sysId, number: req.getValue('u_number') };
        } catch (e) {
            gs.error('HRLXAIAgentTools.requestLeave: ' + e);
            return { ok: false, error: 'exception' };
        }
    },

    checkPolicy: function(question) {
        var profile = gs.getProperty(this.SCOPE + '.ai_search.profile', 'hrlx_kb_profile');
        try {
            var sn = new sn_ais.AISearch();
            var results = sn.search(profile, question, { limit: 5, genai: true });
            return {
                answer: results.genai_answer || '',
                citations: results.citations || []
            };
        } catch (e) {
            return { answer: '', citations: [], error: String(e) };
        }
    },

    escalate: function(userId, summary) {
        var emp = this._getEmployeeByUser(userId);
        if (!emp) return { ok: false, error: 'employee_not_found' };
        var hr = new GlideRecord('sn_hr_core_case');
        if (!hr.isValid()) {
            // Fallback: create audit + notify
            this._audit('ai_escalate_no_hrsd', userId, { summary: summary });
            return { ok: true, case_id: null, note: 'HRSD not installed — audit only' };
        }
        hr.initialize();
        hr.opened_for = userId;
        hr.short_description = 'AI escalation: ' + (summary || '').substring(0, 80);
        hr.description = summary;
        hr.subject_person = userId;
        var sysId = hr.insert();
        this._audit('ai_escalate', sysId, { summary: summary });
        return { ok: true, case_id: sysId };
    },

    getMyRequests: function(userId, days) {
        var emp = this._getEmployeeByUser(userId);
        if (!emp) return [];
        var since = new GlideDateTime();
        since.addDaysLocalTime(-1 * (days || 90));
        var gr = new GlideRecord(this.T_REQ);
        gr.addQuery('u_employee_ref', emp.getUniqueValue());
        gr.addQuery('sys_created_on', '>=', since);
        gr.orderByDesc('sys_created_on');
        gr.query();
        var out = [];
        while (gr.next()) {
            out.push({
                id: gr.getUniqueValue(),
                number: String(gr.u_number),
                type: String(gr.u_leave_type),
                start: String(gr.u_start_date),
                end: String(gr.u_end_date),
                state: String(gr.u_state)
            });
        }
        return out;
    },

    cancelRequest: function(userId, requestId) {
        var gr = new GlideRecord(this.T_REQ);
        if (!gr.get(requestId)) return { ok: false, error: 'not_found' };
        if (gr.u_requested_for_user != userId) return { ok: false, error: 'not_owner' };
        if (gr.u_state != 'submitted') return { ok: false, error: 'not_cancellable' };
        gr.u_state = 'cancelled';
        gr.update();
        this._audit('ai_cancel_request', requestId, { user: userId });
        return { ok: true };
    },

    _audit: function(action, recordId, payload) {
        var a = new GlideRecord(this.T_AUD);
        a.initialize();
        a.u_action = action;
        a.u_record = recordId;
        a.u_payload = JSON.stringify(payload || {});
        a.u_actor = gs.getUserID();
        a.u_source = 'ai_agent';
        a.insert();
    },

    type: 'HRLXAIAgentTools'
};

var HRLXAIProvider = Class.create();
HRLXAIProvider.prototype = {
    initialize: function() {
        this.SCOPE = 'x_786686_hr_leave';
        this.TIMEOUT_MS = parseInt(gs.getProperty(this.SCOPE + '.llm.timeout_ms', '15000'), 10);
    },

    _prop: function(name, def) { return gs.getProperty(this.SCOPE + '.' + name, def); },

    _budgetCheck: function(provider) {
        var budget = parseFloat(this._prop('llm.daily_budget_usd', '10'));
        var key = 'llm.spent.' + provider + '.' + new GlideDateTime().getDate().getDisplayValue();
        return parseFloat(gs.getProperty(this.SCOPE + '.' + key, '0')) < budget;
    },

    askClaude: function(prompt, model, maxTokens, system) {
        if (!this._budgetCheck('anthropic')) return { error: 'budget_exceeded' };
        try {
            var rm = new sn_ws.RESTMessageV2('Claude_API', 'post');
            rm.setHttpTimeout(this.TIMEOUT_MS);
            rm.setStringParameterNoEscape('model', model || this._prop('llm.anthropic.model', 'claude-sonnet-4-5'));
            rm.setStringParameterNoEscape('max_tokens', String(maxTokens || 400));
            rm.setStringParameterNoEscape('system', system || 'You are a helpful HR assistant.');
            rm.setStringParameterNoEscape('user_message', prompt);
            var resp = rm.execute();
            if (resp.getStatusCode() >= 400) return { error: 'http_' + resp.getStatusCode() };
            var json = JSON.parse(resp.getBody());
            var text = '';
            (json.content || []).forEach(function(c) { if (c.type === 'text') text += c.text; });
            return { text: text, usage: json.usage || {} };
        } catch (e) { return { error: 'exception', message: String(e) }; }
    },

    askWithFallback: function(prompt, options) {
        options = options || {};
        var chain = options.chain || ['anthropic', 'openai', 'now_llm'];
        for (var i = 0; i < chain.length; i++) {
            var res = (chain[i] === 'anthropic') ? this.askClaude(prompt, options.model, options.max_tokens, options.system) : null;
            if (res && !res.error && res.text) return res;
        }
        return { error: 'all_providers_failed' };
    },

    type: 'HRLXAIProvider'
};

var HRLXAICache = Class.create();
HRLXAICache.prototype = {
    initialize: function() { this.TABLE = 'x_786686_hr_leave_ai_cache'; },
    _hash: function(s) { return new GlideDigest().getSHA256Base64(s); },
    get: function(skill, input) {
        var gr = new GlideRecord(this.TABLE);
        gr.addQuery('u_hash', this._hash(skill + '|' + input));
        gr.addQuery('u_expires_at', '>', new GlideDateTime());
        gr.setLimit(1);
        gr.query();
        return gr.next() ? String(gr.u_response) : null;
    },
    put: function(skill, input, response, ttlMin) {
        var gr = new GlideRecord(this.TABLE);
        gr.initialize();
        gr.u_hash = this._hash(skill + '|' + input);
        gr.u_skill = skill;
        gr.u_response = response;
        var exp = new GlideDateTime();
        exp.addSeconds(60 * (ttlMin || 60));
        gr.u_expires_at = exp;
        gr.insert();
    },
    type: 'HRLXAICache'
};
