§1 — What is App Engine? / ما هو App Engine؟

Open component:ApplicationLeave Requests

App Engine is ServiceNow’s low-code/no-code platform for citizen developers and pro developers to build custom applications quickly. It bundles:

  • App Engine Studio (AES) — visual app builder
  • Workflow Studio — drag-drop flow designer
  • Mobile Studio — native mobile app builder
  • Service Portal Configurator — quick portal building
  • UI Builder — Workspace pages
  • Now Assist for Creators — AI-assisted app generation — App Engine منصة تطوير منخفض/بدون كود من ServiceNow تتيح للمطورين بناء تطبيقات مخصصة بسرعة.

3.4 Worked example: Visitor Management App

Open component:ApplicationLeave Requests

Step 1 — Create app

AES > Create App > "Visitor Management" > Scope: x_acme_visitor

Step 2 — Add Data table

Table x_acme_visitor_visit:
- first_name (String)
- last_name (String)
- email (Email)
- host_employee (Reference → sys_user)
- visit_date (Date)
- visit_purpose (Choice: Meeting | Interview | Delivery | Tour | Other)
- badge_number (String, auto-numbered)
- check_in_time (Date/Time)
- check_out_time (Date/Time)
- state (Choice: Pending | Approved | Checked In | Checked Out | Cancelled)

Step 3 — Add UI (form layout)

Form sections: Visitor Info | Visit Details | Check-In/Out

Step 4 — Add Logic (Workflow Studio)

Flow: “Visitor Approval”

Trigger: Visitor record created
Step 1: Send email to host with approve/reject link
Step 2: Wait for response (max 4 hours)
Step 3a (approve): Set state = Approved + generate QR badge
Step 3b (reject): Set state = Cancelled + notify visitor
Step 4: Day-of arrival reminder (email + SMS)

Step 5 — Add Experiences

  • Portal page /visitors for self-registration
  • Workspace for reception desk
  • Mobile app for host approvals

Step 6 — Add Connections

  • Active Directory (host validation)
  • Door access system (badge activation)
  • Twilio (SMS)

Step 7 — Publish + share

AES > Publish > Set roles > Add to Application Repository

17.3 Top 15 App Engine Scripts

Open component:ApplicationLeave Requests

1. AES app inventory + compliance scanner

var AESComplianceScanner = Class.create();
AESComplianceScanner.prototype = {
 initialize: function {},

/* Scans all citizen-built apps and rates compliance /
scan: function {
var apps = new GlideRecord(‘sys_app’);
apps.addQuery(‘source’, ‘app_engine_studio’);
apps.addActiveQuery();
apps.query();
var results = [];
while (apps.next()) {
var report = this._auditApp(apps.getUniqueValue());
results.push({
app: apps.getValue(‘name’),
scope: apps.getValue(‘scope’),
score: report.score,
issues: report.issues
});
}
return results;
},

_auditApp: function (appSysId) {
var score = 100;
var issues = [];

// Check 1: explicit roles defined
var roles = new GlideAggregate(‘sys_user_role’);
roles.addQuery(‘sys_scope’, appSysId);
roles.addAggregate(‘COUNT’);
roles.query();
if (roles.next() && parseInt(roles.getAggregate(‘COUNT’), 10) === 0) {
score -= 20;
issues.push(‘no_explicit_roles’);
}

// Check 2: ACLs defined
var acls = new GlideAggregate(‘sys_security_acl’);
acls.addQuery(‘sys_scope’, appSysId);
acls.addAggregate(‘COUNT’);
acls.query();
if (acls.next() && parseInt(acls.getAggregate(‘COUNT’), 10) < 3) {
score -= 15;
issues.push(‘insufficient_acls’);
}

// Check 3: ATF tests exist
var atf = new GlideAggregate(‘sys_atf_test’);
atf.addQuery(‘sys_scope’, appSysId);
atf.addAggregate(‘COUNT’);
atf.query();
if (atf.next() && parseInt(atf.getAggregate(‘COUNT’), 10) === 0) {
score -= 25;
issues.push(‘no_atf_tests’);
}

return { score: Math.max(0, score), issues: issues };
},

type: ‘AESComplianceScanner’
};

2. App-level role provisioning

function provisionAppRoles(appScope, roleNames, userSysIds) {
 roleNames.forEach(function (roleName) {
 var roleId = _ensureRole(appScope, roleName);
 userSysIds.forEach(function (userId) {
 var has = new GlideRecord('sys_user_has_role');
 has.addQuery('user', userId);
 has.addQuery('role', roleId);
 has.setLimit(1);
 has.query();
 if (!has.hasNext()) {
 has.initialize();
 has.setValue('user', userId);
 has.setValue('role', roleId);
 has.insert();
 }
 });
 });

function _ensureRole(scope, name) {
var r = new GlideRecord(‘sys_user_role’);
r.addQuery(‘name’, scope + ‘.’ + name);
r.setLimit(1); r.query();
if (r.next()) return r.getUniqueValue();
r.initialize();
r.setValue(‘name’, scope + ‘.’ + name);
r.setValue(‘description’, ‘Auto-provisioned role for ’ + scope);
return r.insert();
}
}
