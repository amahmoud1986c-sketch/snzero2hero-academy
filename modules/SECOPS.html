4.2 Vulnerability prioritization

Open component:ApplicationLeave Requests

Default formula: CVSS × Asset Criticality × Exploit Availability × Business Impact

var VRPrioritizer = Class.create();
VRPrioritizer.prototype = {
 initialize: function {},

/* Computes priority 1 (Critical) to 5 (Info) /
prioritize: function (viSysId) {
var vi = new GlideRecord(‘sn_vul_vulnerable_item’);
if (!vi.get(viSysId)) return 5;
var cvss = parseFloat(vi.getValue(‘vulnerability.base_score’)) || 0;
var ciCrit = this._ciCriticality(vi.getValue(‘cmdb_ci’));
var exploit = this._exploitAvailable(vi.getValue(‘vulnerability.cve_id’));
var biz = this._businessImpact(vi.getValue(‘cmdb_ci’));

var score = (cvss * 0.4) + (ciCrit * 0.25) + (exploit * 0.2) + (biz * 0.15);
var prio = score >= 8 ? 1 : (score >= 6 ? 2 : (score >= 4 ? 3 :
(score >= 2 ? 4 : 5)));

vi.setValue(‘risk_score’, score.toFixed(1));
vi.setValue(‘priority’, prio);
vi.update();
return prio;
},

_ciCriticality: function (ciId) {
var c = new GlideRecord(‘cmdb_ci’);
if (!c.get(ciId)) return 5;
var bizCrit = c.getValue(‘business_criticality’);
var map = { ‘1 - most critical’: 10, ‘2 - somewhat critical’: 7,
‘3 - less critical’: 4, ‘4 - not critical’: 2 };
return map[bizCrit] || 5;
},

_exploitAvailable: function (cve) {
var fb = new GlideRecord(‘sn_vul_third_party_entry’);
fb.addQuery(‘cve_id’, cve);
fb.addQuery(‘exploit_status’, ‘available’);
fb.setLimit(1);
fb.query();
return fb.hasNext() ? 10 : 0;
},

_businessImpact: function (ciId) {
var s = new GlideAggregate(‘cmdb_ci_service_auto’);
s.addQuery(‘cmdb_ci_dependents’, ciId);
s.addAggregate(‘COUNT’);
s.query();
if (s.next()) {
var n = parseInt(s.getAggregate(‘COUNT’), 10);
return Math.min(10, n);
}
return 0;
},

type: ‘VRPrioritizer’
};

// Group VIs by patch ID → bulk Change
(function bulkChange() {
 var agg = new GlideAggregate('sn_vul_vulnerable_item');
 agg.addQuery('state', 'open');
 agg.addQuery('priority', 'IN', '1,2');
 agg.addAggregate('COUNT');
 agg.groupBy('vulnerability.patch_id');
 agg.query();

while (agg.next()) {
var patchId = agg.getValue(‘vulnerability.patch_id’);
if (!patchId) continue;
var count = parseInt(agg.getAggregate(‘COUNT’), 10);
if (count >= 5) {
var chg = new GlideRecord(‘change_request’);
chg.initialize();
chg.setValue(‘short_description’, ‘Apply patch ’ + patchId +
’ to ’ + count + ’ affected systems’);
chg.setValue(‘category’, ‘security’);
chg.setValue(‘priority’, 2);
chg.setValue(‘risk’, ‘moderate’);
chg.insert();
}
}
})();

5.4 Threat hunt: IoC sweep across the estate

Open component:ApplicationLeave Requests
function huntIoC(observableValue, observableType) {
 var findings = [];

if (observableType === ‘ip’) {
// Search active sessions / firewall logs / SIEM
var siem = new sn_ws.RESTMessageV2(‘Splunk’, ‘searchIoC’);
siem.setStringParameter(‘ioc’, observableValue);
siem.setHttpTimeout(60000);
var resp = JSON.parse(siem.execute().getBody() || ‘{}’);
if (resp.results && resp.results.length > 0) {
resp.results.forEach(function (hit) {
findings.push({ source: ‘siem’, host: hit.host, time: hit.time });
});
}
} else if (observableType === ‘hash’) {
// Check EDR for file presence
var edr = new sn_ws.RESTMessageV2(‘CrowdStrikeEDR’, ‘searchHash’);
edr.setStringParameter(‘hash’, observableValue);
var edrResp = JSON.parse(edr.execute().getBody() || ‘{}’);
(edrResp.devices || []).forEach(function (d) {
findings.push({ source: ‘edr’, host: d.hostname, time: d.last_seen });
});
}

return findings;
}

(function process(request, response) {
 var alert = request.body.data;
 var inc = new GlideRecord('sn_si_incident');
 inc.initialize();
 inc.setValue('short_description', alert.title);
 inc.setValue('description', alert.description);
 inc.setValue('severity', _mapSeverity(alert.severity));
 inc.setValue('source', alert.source || 'siem');
 inc.setValue('correlation_id', alert.alert_id);
 var sysId = inc.insert();

// Affected CIs
(alert.affected_hosts || []).forEach(function (host) {
var ci = new GlideRecord(‘cmdb_ci’);
ci.addQuery(‘name’, host);
ci.setLimit(1); ci.query();
if (ci.next()) {
var m = new GlideRecord(‘sn_si_m2m_ci_incident’);
m.initialize();
m.setValue(‘incident’, sysId);
m.setValue(‘configuration_item’, ci.getUniqueValue());
m.insert();
}
});

// IoCs
(alert.iocs || []).forEach(function (ioc) {
var ob = new GlideRecord(‘sn_ti_observable’);
ob.initialize();
ob.setValue(‘value’, ioc.value);
ob.setValue(‘finding’, ioc.type);
ob.setValue(‘source’, ‘incident_’ + inc.getValue(‘number’));
ob.insert();
});

response.setStatus(201);
return { incident_id: inc.getValue(‘number’) };

function _mapSeverity(sev) {
var m = { critical: 1, high: 2, medium: 3, low: 4, info: 5 };
return m[String(sev).toLowerCase()] || 3;

function blockIP(ip, reason, ticketId) {
 var rm = new sn_ws.RESTMessageV2('PaloAltoFirewall', 'addBlocklist');
 rm.setStringParameter('ip', ip);
 rm.setStringParameter('note', 'SIR-' + ticketId + ': ' + reason);
 rm.setHttpTimeout(20000);
 try {
 var r = rm.execute();
 if (r.getStatusCode() >= 200 && r.getStatusCode() < 300) {
 gs.eventQueue('x_sn_si.ip.blocked', null, ip, ticketId);
 return true;

 gs.error('[SIR] FW block failed: ' + r.getBody());

 return false;
}
