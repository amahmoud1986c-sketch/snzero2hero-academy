15.2 What is IRM vs GRC vs BCM vs TPRM? (business language)

Open component:RisksControlsIssuesAudit Engagements

Full map with module boundaries: §1B GRC vs IRM vs Modules

GRC — industry umbrella (Governance + Risk + Compliance). — GRC = مجال الأعمال — ليس اسم تطبيق ServiceNow فقط.

IRM — ServiceNow’s integrated platform for GRC (multiple apps, shared Profiles/Issues). — IRM = تنفيذ GRC على ServiceNow — عدة تطبيقات مترابطة.

BCM — Business Continuity: BIA, recovery plans, exercises (after disruption). — BCM = استمرارية الأعمال — BIA + خطط + تمارين.

TPRM — Third-Party Risk: vendors, SaaS, outsourcers (own app: Vendor Risk / VRM). — TPRM = مخاطر الموردين — تطبيق Vendor Risk منفصل عن Risk الداخلي.

GRC (discipline — what client wants)
 │
 ▼
IRM (ServiceNow suite)
├─ Policy & Compliance → controls, attestations, indicators ← "Am I compliant?"
├─ Risk Management → register, assess, treat ← "What could go wrong?"
├─ Audit Management → engagements, findings ← "Can auditor verify?"
├─ TPRM (Vendor Risk) → tier, assess vendors ← "Are suppliers safe?"
└─ BCM → BIA, plans, exercises ← "Can we recover?"

Do not confuse: Compliance module ≠ Risk module ≠ Audit module — separate licenses, linked by Profile and Issue. — كل موديول ترخيص وقوائم مختلفة — يربطهم Profile و Issue.

1. Auto-create issue when indicator fails

// Business Rule on sn_grc_indicator (after, update)
(function (current, previous) {
 if (current.last_result.changes() && current.getValue('last_result') === 'fail') {
 var issue = new GlideRecord('sn_grc_issue');
 issue.initialize();
 issue.setValue('short_description', 'Indicator failure: ' + current.getValue('name'));
 issue.setValue('description', 'Indicator ' + current.getValue('number') +
 ' returned failure. Investigate root cause and remediate.');
 issue.setValue('source_indicator', current.getUniqueValue());
 issue.setValue('control', current.getValue('control'));
 issue.setValue('priority', _priority(current));
 issue.setValue('owner', current.getValue('control.owned_by'));
 issue.insert();
 }

function _priority(ind) {
var ctrl = new GlideRecord(‘sn_compliance_control’);
if (!ctrl.get(ind.getValue(‘control’))) return 3;
return ctrl.getValue(‘classification’) === ‘key’ ? 1 : 3;
}
})(current, previous);

2. Vendor risk score calculator

var GRCVendorScorer = Class.create();
GRCVendorScorer.prototype = {
 initialize: function {},

/* Computes vendor inherent risk based on data access, criticality, location /
score: function (vendorSysId) {
var v = new GlideRecord(‘sn_vdr_risk_asmt_vendor’);
if (!v.get(vendorSysId)) return 0;

var dataSensitivity = parseInt(v.getValue(‘data_sensitivity’), 10) || 0; // 1-5
var businessCrit = parseInt(v.getValue(‘business_criticality’), 10) || 0;
var geoRisk = this._geoRisk(v.getValue(‘country’));
var spendTier = this._spendTier(parseFloat(v.getValue(‘annual_spend’)));

var score = (dataSensitivity * 0.35) + (businessCrit * 0.30) +
(geoRisk * 0.20) + (spendTier * 0.15);
var tier = score >= 4 ? 1 : (score >= 2.5 ? 2 : 3);

v.setValue(‘inherent_risk_score’, score.toFixed(2));
v.setValue(‘tier’, tier);
v.update();
return { score: score, tier: tier };
},

_geoRisk: function (cc) {
var high = [‘AF’, ‘IR’, ‘KP’, ‘SY’, ‘YE’];
var med = [‘IN’, ‘CN’, ‘RU’, ‘BR’];
if (high.indexOf(cc) > -1) return 5;
if (med.indexOf(cc) > -1) return 3;
return 1;
},

_spendTier: function (amt) {
if (amt >= 1000000) return 5;
if (amt >= 100000) return 3;
return 1;
},

type: ‘GRCVendorScorer’
};

function launchAttestationCampaign(frameworkSysId, dueDate) {
 var ctrl = new GlideRecord('sn_compliance_control');
 ctrl.addQuery('content.framework', frameworkSysId);
 ctrl.addActiveQuery();
 ctrl.query();

var count = 0;
while (ctrl.next()) {
var att = new GlideRecord(‘sn_compliance_control_attestation’);
att.initialize();
att.setValue(‘control’, ctrl.getUniqueValue());
att.setValue(‘attestor’, ctrl.getValue(‘owned_by’));
att.setValue(‘due_date’, dueDate);
att.setValue(‘state’, ‘pending’);
att.insert();
count++;
}
return { campaign: frameworkSysId, count: count };
}

function auditTrail(issueSysId) {
 var trail = [];
 var i = new GlideRecord('sn_grc_issue');
 if (!i.get(issueSysId)) return null;
 trail.push({ type: 'issue', id: issueSysId, num: i.getValue('number') });

if (!i.control.nil()) {
var c = new GlideRecord(‘sn_compliance_control’);
if (c.get(i.getValue(‘control’))) {
trail.push({ type: ‘control’, id: c.getUniqueValue(),
num: c.getValue(‘number’) });
var prof = new GlideRecord(‘sn_grc_profile’);
if (prof.get(c.getValue(‘profile’))) {
trail.push({ type: ‘profile’, id: prof.getUniqueValue(),
name: prof.getValue(‘name’) });
}
}
}

if (!i.risk.nil()) {
var r = new GlideRecord(‘sn_risk_risk’);
if (r.get(i.getValue(‘risk’))) {
trail.push({ type: ‘risk’, id: r.getUniqueValue(),
num: r.getValue(‘number’) });
}
}
return trail;
}

6. Residual risk = Inherent × (1 - control_effectiveness)

var GRCRiskCalculator = Class.create();
GRCRiskCalculator.prototype = {
 initialize: function {},

calcResidual: function (riskSysId) {
var r = new GlideRecord(‘sn_risk_risk’);
if (!r.get(riskSysId)) return 0;
var inherent = parseFloat(r.getValue(‘inherent_score’));
var ctrlEff = this._linkedControlsEffectiveness(riskSysId);
var residual = inherent * (1 - ctrlEff);
r.setValue(‘residual_score’, residual.toFixed(2));
r.update();
return residual;
},

_linkedControlsEffectiveness: function (riskId) {
var link = new GlideAggregate(‘sn_risk_risk_control’);
link.addQuery(‘risk’, riskId);
link.addAggregate(‘AVG’, ‘control.effectiveness_score’);
link.query();
return link.next() ?
parseFloat(link.getAggregate(‘AVG’, ‘control.effectiveness_score’)) / 100 : 0;
},

type: ‘GRCRiskCalculator’
};
