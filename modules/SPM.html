1) Overview & Business Value

Open component:DemandsProjectsPortfoliosResource Plans

SPM (Strategic Portfolio Management) is ServiceNow’s suite for turning strategy into funded, resourced, delivered work.
PPM (Project Portfolio Management) is the project/delivery app inside that suite — not a rival to SPM. — SPM = مجموعة تطبيقات (استراتيجية → فكرة → طلب → مشروع → موارد → ميزانية). PPM = جزء من SPM يُدير المشاريع والتسليم — ليس بديلًا لـ SPM.

Read first: §1B Strategy → Goals → Innovation → Demand → Project — Full Map

Executives set strategy & goals; the organization submits ideas; PMO governs demands; approved work becomes projects with resources and financials — all visible on portfolios and roadmaps.

الإدارة تضع الاستراتيجية والأهداف → الموظفون يقدّمون أفكارًا → PMO يقيّم الطلبات → المعتمد يصبح مشروعًا → موارد وميزانية → منافع تُغذّي الاستراتيجية مجددًا.

1B.10 Quick interview / workshop answers

Open component:DemandsProjectsPortfoliosResource Plans
Question Short answer (EN) العربي

9) Best Practices & Common Mistakes

Open component:DemandsProjectsPortfoliosResource Plans

Best practices:
- Start with Demand to enforce intake governance.
- Use Planning Console for scheduling (not manual dates).
- Keep resource plans realistic (capacity vs allocation).
- Tie projects to portfolios for strategic visibility.

Common mistakes:
- إنشاء مشاريع بلا Demand (لا حوكمة دخول).
- تجاهل Resource capacity فتتعارض التخصيصات.
- خلط Program و Portfolio.

2. Demand scoring formula

var SPMDemandScorer = Class.create();
SPMDemandScorer.prototype = {
 initialize: function {},

/* Computes weighted score: strategic 40%, financial 30%, risk 20%, effort 10% /
score: function (demandSysId) {
var d = new GlideRecord(‘dmn_demand’);
if (!d.get(demandSysId)) return 0;

var strategic = this._n(d.getValue(‘strategic_alignment’));
var financial = this._npv(d);
var risk = 100 - this._n(d.getValue(‘risk_score’));
var effort = 100 - this._n(d.getValue(‘effort_score’));

var weighted = (strategic * 0.4) + (financial * 0.3) +
(risk * 0.2) + (effort * 0.1);
var rounded = Math.round(weighted);

d.setValue(‘overall_score’, rounded);
d.update();
return rounded;
},

_n: function (v) { return parseFloat(v || 0); },

_npv: function (d) {
var benefit = parseFloat(d.getValue(‘expected_benefit’) || 0);
var cost = parseFloat(d.getValue(‘expected_cost’) || 0);
if (cost <= 0) return 0;
return Math.min(100, Math.round((benefit - cost) / cost * 100));
},

type: ‘SPMDemandScorer’
};

3. Resource over-allocation detector

function detectOverAllocations(periodStart, periodEnd) {
 var conflicts = [];
 var resources = new GlideRecord('sys_user');
 resources.addActiveQuery();
 resources.addQuery('u_resource', true);
 resources.query();
 while (resources.next()) {
 var capacity = parseFloat(resources.getValue('u_weekly_capacity') || 40);
 var alloc = new GlideAggregate('resource_allocation');
 alloc.addQuery('resource', resources.getUniqueValue());
 alloc.addQuery('start_date', '<=', periodEnd);
 alloc.addQuery('end_date', '>=', periodStart);
 alloc.addAggregate('SUM', 'hours');
 alloc.query();
 if (alloc.next()) {
 var totalH = parseFloat(alloc.getAggregate('SUM', 'hours'));
 if (totalH > capacity) {
 conflicts.push({
 user: resources.getDisplayValue(),
 capacity: capacity,
 allocated: totalH,
 over: totalH - capacity
 });
 }
 }
 }
 return conflicts;
}

function burnDown(sprintSysId) {
 var s = new GlideRecord('rm_sprint');
 if (!s.get(sprintSysId)) return null;
 var start = new GlideDateTime(s.getValue('start_date'));
 var end = new GlideDateTime(s.getValue('end_date'));
 var totalPoints = 0;

var st = new GlideRecord(‘rm_story’);
st.addQuery(‘sprint’, sprintSysId);
st.query();
while (st.next()) {
totalPoints += parseInt(st.getValue(‘story_points’), 10) || 0;
}

var days = [];
var d = new GlideDateTime(start);
while (d.compareTo(end) <= 0) {
var done = new GlideAggregate(‘rm_story’);
done.addQuery(‘sprint’, sprintSysId);
done.addQuery(‘state’, ‘done’);
done.addQuery(‘sys_updated_on’, ‘<=’, d);
done.addAggregate(‘SUM’, ‘story_points’);
done.query();
var burned = done.next() ? parseInt(done.getAggregate(‘SUM’, ‘story_points’), 10) : 0;
days.push({ date: d.getDate().getValue(), remaining: totalPoints - burned });
d.addDaysLocalTime(1);
}
return { total: totalPoints, days: days };
}

6. Status RAG calculator

var SPMRAGCalculator = Class.create();
SPMRAGCalculator.prototype = {
 initialize: function {},

/* Returns ‘green’ | ‘amber’ | ‘red’ /
calc: function (projectSysId) {
var p = new GlideRecord(‘pm_project’);
if (!p.get(projectSysId)) return ‘unknown’;
var schedule = this._schedule(p);
var cost = this._cost(p);
var risk = this._risk(p);

var worst = Math.max(schedule, cost, risk);
if (worst >= 2) return ‘red’;
if (worst >= 1) return ‘amber’;
return ‘green’;
},

_schedule: function (p) {
var planned = new GlideDateTime(p.getValue(‘planned_end_date’));
var current = new GlideDateTime(p.getValue(‘end_date’));
var diffDays = (current.getNumericValue() - planned.getNumericValue()) /
(1000 * 60 * 60 * 24);
if (diffDays > 14) return 2;
if (diffDays > 0) return 1;
return 0;
},

_cost: function (p) {
var plan = parseFloat(p.getValue(‘budget_cost’) || 0);
var act = parseFloat(p.getValue(‘actual_cost’) || 0);
if (plan <= 0) return 0;
var over = (act - plan) / plan;
if (over > 0.1) return 2;
if (over > 0) return 1;
return 0;
},

_risk: function (p) {
var r = new GlideAggregate(‘pm_project_risk’);
r.addQuery(‘project’, p.getUniqueValue());
r.addQuery(‘state’, ‘!=’, ‘closed’);
r.addQuery(‘severity’, ‘IN’, ‘4,5’); // high/critical
r.addAggregate(‘COUNT’);
r.query();
var critRisks = r.next() ? parseInt(r.getAggregate(‘COUNT’), 10) : 0;
if (critRisks > 3) return 2;
if (critRisks > 0) return 1;
return 0;
},

type: ‘SPMRAGCalculator’
};
