1) Overview & Business Value

Open component:IncidentsProblemsChangesSOW

ITSM manages the full lifecycle of IT services for employees. — هو الموديول الأساسي الذي تبدأ منه معظم تطبيقات ServiceNow.

Core processes (ITIL 4 aligned):
- Incident Management — استعادة الخدمة بسرعة بعد عطل.
- Problem Management — معالجة السبب الجذري لمنع التكرار.
- Change Management — تغيير الخدمات بأمان ومخاطر محسوبة.
- Request Management — تلبية الطلبات عبر الكتالوج.
- Service Level Management (SLA) — قياس الالتزام بالأوقات.
- Knowledge Management — توثيق الحلول ونشر المعرفة.

9) Best Practices & Common Mistakes

Open component:IncidentsProblemsChangesSOW

Best practices:
- Use Flow Designer, not legacy Workflow, for new approvals.
- Keep priority via Impact×Urgency matrix (don’t hardcode).
- Use Assignment Rules / Data Lookup instead of long scripts.
- Always relate the CI (cmdb_ci) for reporting and impact.

Common mistakes:
- خلط Standard و Normal change.
- نسيان شرط Stop في الـ SLA فيبقى يعمل.
- كتابة منطق إسناد في Business Rule بدل Assignment Rule.
- تعديل جداول OOTB بدل الإضافة عبر scope.

// Business Rule: before insert on change_request — set risk score
(function riskScore() {
 var score = 0;
 if (current.type == 'emergency') score += 30;
 if (current.cmdb_ci && current.cmdb_ci.business_criticality == '1') score += 20;
 if (gs.daysBetween(current.start_date, current.end_date) > 1) score += 10;
 var hasBackout = current.backout_plan && current.backout_plan.length > 50;
 if (!hasBackout) score += 25;
 var hasTestPlan = current.test_plan && current.test_plan.length > 50;
 if (!hasTestPlan) score += 15;

current.risk = score >= 70 ? ‘high’ : (score >= 40 ? ‘moderate’ : ‘low’);
current.risk_value = score;
})();

Wednesday — conflict detection (via change_conflict):

// Script Action — find conflicts before CAB
(function findChangeConflict(chg) {
 var conf = new GlideRecord('change_request');
 conf.addQuery('cmdb_ci', chg.cmdb_ci);
 conf.addQuery('sys_id', '!=', chg.sys_id);
 conf.addQuery('start_date', '<=', chg.end_date);
 conf.addQuery('end_date', '>=', chg.start_date);
 conf.addQuery('state', 'IN', '-2,-3,-4'); // Assess, Authorize, Scheduled
 conf.query();
 var conflicts = [];
 while (conf.next()) {
 conflicts.push(conf.number + ' on ' + conf.cmdb_ci.name);
 // Create change_conflict record
 var c = new GlideRecord('change_conflict');
 c.initialize();
 c.change = chg.sys_id;
 c.conflicting_change = conf.sys_id;
 c.reason = 'Overlapping window on same CI';
 c.insert();
 }
 if (conflicts.length) {
 chg.work_notes = 'CONFLICTS DETECTED: ' + conflicts.join(', ');
 chg.update();
 }
})(current);

6) CI relationship traverser — blast radius

// Script Include or Background script — find downstream CIs
function findBlastRadius(rootCiSysId, maxDepth) {
 var visited = {};
 var queue = [{id: rootCiSysId, depth: 0}];
 var result = [];
 while (queue.length) {
 var node = queue.shift();
 if (visited[node.id] || node.depth > (maxDepth || 5)) continue;
 visited[node.id] = true;
 result.push(node.id);
 var rel = new GlideRecord('cmdb_rel_ci');
 rel.addQuery('parent', node.id);
 rel.query();
 while (rel.next()) {
 queue.push({id: rel.child.toString(), depth: node.depth + 1});
 }
 }
 return result;
}
// Usage:
var blast = findBlastRadius('<ci_sys_id>', 4);
gs.info('Blast radius: ' + blast.length + ' downstream CIs');
