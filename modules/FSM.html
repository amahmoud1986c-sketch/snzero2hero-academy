1. Auto-assign WO by skill + territory + availability

var FSMSmartDispatch = Class.create();
FSMSmartDispatch.prototype = {
 initialize: function {},

assign: function (woSysId) {
var wo = new GlideRecord(‘wm_order’);
if (!wo.get(woSysId)) return null;
var requiredSkill = wo.getValue(‘skills’);
var territory = wo.getValue(‘territory’);
var due = wo.getValue(‘requested_due_by’);

// candidates with skill + territory
var cand = new GlideRecord(‘sys_user’);
cand.addQuery(‘active’, true);
cand.addQuery(‘skills’, ‘CONTAINS’, requiredSkill);
cand.addQuery(‘u_territories’, ‘CONTAINS’, territory);
cand.query();

var best = null;
var bestScore = -Infinity;
while (cand.next()) {
var score = this._score(cand, wo);
if (score > bestScore) {
bestScore = score;
best = cand.getUniqueValue();
}
}
if (best) {
wo.setValue(‘assigned_to’, best);
wo.update();
}
return best;
},

_score: function (tech, wo) {
// Lower load + nearby + matching language
var loadAgg = new GlideAggregate(‘wm_order’);
loadAgg.addQuery(‘assigned_to’, tech.getUniqueValue());
loadAgg.addQuery(‘state’, ‘IN’, ‘pending,in_progress’);
loadAgg.addAggregate(‘COUNT’);
loadAgg.query();
var load = loadAgg.next() ? parseInt(loadAgg.getAggregate(‘COUNT’), 10) : 0;
return -load * 10; // negative = lower is better
},

type: ‘FSMSmartDispatch’
};

2. Route optimization (simple greedy)

function optimizeDay(techSysId, date) {
 var gr = new GlideRecord('wm_order');
 gr.addQuery('assigned_to', techSysId);
 gr.addQuery('scheduled_start', date);
 gr.orderBy('priority');
 gr.query();

var wos = [];
while (gr.next()) {
wos.push({
sysId: gr.getUniqueValue(),
lat: parseFloat(gr.getValue(‘latitude’)),
lng: parseFloat(gr.getValue(‘longitude’)),
duration: parseInt(gr.getValue(‘estimated_duration’), 10),
priority: parseInt(gr.getValue(‘priority’), 10)
});
}

// Nearest-neighbor heuristic from start point
var start = { lat: parseFloat(gs.getProperty(‘x_fsm.depot_lat’)),
lng: parseFloat(gs.getProperty(‘x_fsm.depot_lng’)) };
var ordered = [];
var cur = start;
while (wos.length > 0) {
wos.sort(function (a, b) {
return _dist(cur, a) - _dist(cur, b);
});
var next = wos.shift();
ordered.push(next);
cur = next;
}

// Persist order
ordered.forEach(function (w, idx) {
var u = new GlideRecord(‘wm_order’);
if (u.get(w.sysId)) {
u.setValue(‘dispatch_order’, idx + 1);
u.update();
}
});

function _dist(a, b) {
var dx = a.lat - b.lat, dy = a.lng - b.lng;
return Math.sqrt(dx * dx + dy * dy);
}
}

for (var i = 0; i < partRequests.length; i++) {
var p = partRequests[i];
var stock = new GlideRecord(‘alm_consumable’);
stock.addQuery(‘stockroom’, truck);
stock.addQuery(‘model’, p.model);
stock.setLimit(1);
stock.query();
if (!stock.next()) {
return { success: false, error: ‘no_stock’, model: p.model };
}
var avail = parseInt(stock.getValue(‘quantity’), 10) -
parseInt(stock.getValue(‘reserved’), 10);
if (avail < p.qty) {
return { success: false, error: ‘insufficient’, model: p.model,
available: avail, requested: p.qty };
}
stock.setValue(‘reserved’,
parseInt(stock.getValue(‘reserved’), 10) + p.qty);
stock.update();
reserved.push({ stock: stock.getUniqueValue(), qty: p.qty });
}

return { success: true, reserved: reserved };
}

5. Geofence arrival detection

// Scripted REST endpoint /api/x_fsm/geofence
(function process(request, response) {
 var data = request.body.data;
 var techId = data.tech_id;
 var lat = parseFloat(data.lat);
 var lng = parseFloat(data.lng);

// Find active WO for this tech
var wo = new GlideRecord(‘wm_order’);
wo.addQuery(‘assigned_to’, techId);
wo.addQuery(‘state’, ‘en_route’);
wo.setLimit(1);
wo.query();
if (!wo.next()) {
response.setStatus(404); return { ok: false };
}

var siteLat = parseFloat(wo.getValue(‘latitude’));
var siteLng = parseFloat(wo.getValue(‘longitude’));
var distM = _haversine(lat, lng, siteLat, siteLng);

if (distM < 100) {
wo.setValue(‘state’, ‘on_site’);
wo.setValue(‘arrived_at’, new GlideDateTime());
wo.update();
gs.eventQueue(‘x_fsm.wo.arrived’, wo, techId);
}
return { ok: true, distance_m: distM };

function _haversine(lat1, lon1, lat2, lon2) {
var R = 6371000;
var dLat = (lat2 - lat1) * Math.PI / 180;
var dLon = (lon2 - lon1) * Math.PI / 180;
var a = Math.sin(dLat/2) * Math.sin(dLat/2) +
Math.cos(lat1 * Math.PI/180) * Math.cos(lat2 * Math.PI/180) *
Math.sin(dLon/2) * Math.sin(dLon/2);
return 2 * R * Math.atan2(Math.sqrt(a), Math.sqrt(1-a));
}
})(request, response);
