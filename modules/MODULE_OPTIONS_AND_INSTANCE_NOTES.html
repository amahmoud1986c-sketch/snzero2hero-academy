// Run in Background Scripts to enumerate active plugins per scope
(function {
 var plugins = new GlideRecord('sys_plugin');
 plugins.addQuery('active', true);
 plugins.orderBy('name');
 plugins.query();
 var list = [];
 while (plugins.next()) {
 list.push(plugins.getValue('source') + '\t' + plugins.getValue('name'));
 }
 gs.info('[Plugins] active=' + list.length);
 list.forEach(function (l) { gs.info(l); });
})();
