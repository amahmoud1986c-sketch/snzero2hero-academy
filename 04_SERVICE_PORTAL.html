Full script — hrlx_my_leaves Client script

/**
 * Widget: hrlx_my_leaves — Client Controller
 * Application: Custom · Scope: x_786686_hr_leave · sys_id: 83d5298c932b8f107520703efaba1072
 */
function HrlxMyLeavesCtrl() {
    var c = this;

    c.cancelLeave = function (item) {
        if (!confirm('Cancel leave ' + item.number + '?')) return;
        c.data.action = 'cancel';
        c.data.sys_id = item.sys_id;
        c.server.update().then(function () {
            c.data.action = '';
            c.data.sys_id = '';
        });
    };
}

Full script — hrlx_my_leaves Server script

/**
 * Widget: hrlx_my_leaves — Server Script
 * Application: Custom · Scope: x_786686_hr_leave · sys_id: 83d5298c932b8f107520703efaba1072
 * Shows the current user's leave requests (secure).
 */
(function () {
    data.leaves = [];

    var gr = new GlideRecordSecure('x_786686_hr_leave_req_v2');
    gr.addQuery('u_requested_for_user', gs.getUserID());
    gr.orderByDesc('sys_created_on');
    gr.setLimit(50);
    gr.query();

    while (gr.next()) {
        data.leaves.push({
            sys_id: gr.getUniqueValue(),
            number: gr.getValue('u_number'),
            type: gr.getDisplayValue('u_leave_type'),
            start: gr.getValue('u_start_date'),
            end: gr.getValue('u_end_date'),
            days: gr.getValue('u_days_count'),
            state: gr.getValue('u_state')
        });
    }

    // Cancel action from client
    if (input && input.action === 'cancel' && input.sys_id) {
        var c = new GlideRecord('x_786686_hr_leave_req_v2');
        if (c.get(input.sys_id) && c.getValue('u_requested_for_user') === gs.getUserID()) {
            c.u_state = 'cancelled';
            c.update();
        }
    }
})();

Full script — hrlx_balance Server script

/**
 * Widget: hrlx_balance — Server Script
 * Application: Custom · Scope: x_786686_hr_leave · sys_id: 83d5298c932b8f107520703efaba1072
 * Shows the current user's remaining leave balance for the current year.
 */
(function () {
    data.balances = [];
    data.year = new GlideDateTime().getYearLocalTime();

    // find employee record of current user
    var emp = new GlideRecord('x_786686_hr_leave_employee_v2');
    emp.addQuery('u_user', gs.getUserID());
    emp.query();
    if (!emp.next()) return;

    var bal = new GlideRecordSecure('x_786686_hr_leave_bal_v2');
    bal.addQuery('u_employee_ref', emp.getUniqueValue());
    bal.addQuery('u_year', data.year);
    bal.query();
    while (bal.next()) {
        data.balances.push({
            type: bal.getDisplayValue('u_leave_type'),
            entitled: parseInt(bal.getValue('u_entitled') || 0, 10),
            used: parseInt(bal.getValue('u_used') || 0, 10),
            remaining: parseInt(bal.getValue('u_remaining') || 0, 10)
        });
    }
})();
