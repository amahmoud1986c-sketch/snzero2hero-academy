Client Controller:

api.controller = function($scope, spUtil, spModal) {
  var c = this;

  c.request = {
    leave_type: '',
    start_date: null,
    end_date: null,
    reason: '',
    duration: 0
  };

  c.submitting = false;

  // Calculate duration when dates change
  c.calculateDuration = function() {
    if (c.request.start_date && c.request.end_date) {
      var start = new Date(c.request.start_date);
      var end = new Date(c.request.end_date);

      if (end >= start) {
        var diff = Math.ceil((end - start) / (1000 * 60 * 60 * 24)) + 1;
        c.request.duration = diff;
      } else {
        c.request.duration = 0;
        spUtil.addErrorMessage('End date must be after start date');
      }
    }
  };

  // Submit request
  c.submitRequest = function() {
    if (!c.request.leave_type || !c.request.start_date || !c.request.end_date) {
      spUtil.addErrorMessage('Please fill all required fields');
      return;
    }

    if (c.request.duration > c.data.balance) {
      spUtil.addErrorMessage('Request exceeds available balance');
      return;
    }

    c.submitting = true;

    c.server.get({
      action: 'submitRequest',
      request: c.request
    }).then(function(response) {
      c.submitting = false;
      if (response.data.success) {
        c.data.success = true;
        c.data.newRequest = response.data.newRequest;
        spUtil.addInfoMessage('Leave request submitted successfully!');
      } else {
        spUtil.addErrorMessage(response.data.error || 'Failed to submit');
      }
    });
  };

  // Refresh balance when leave type changes
  $scope.$watch('c.request.leave_type', function(newVal) {
    if (newVal) {
      c.server.get({
        action: 'getBalance',
        leave_type: newVal
      }).then(function(response) {
        c.data.balance = response.data.balance;
      });
    }
  });
};

Server Script:

(function() {

  // Get current user info
  data.employee = {
    sys_id: gs.getUserID(),
    name: gs.getUserDisplayName(),
    email: gs.getUser().getEmail()
  };

  // Get leave types
  data.leaveTypes = [
    { value: 'annual', label: 'Annual Leave - إجازة سنوية' },
    { value: 'sick', label: 'Sick Leave - إجازة مرضية' },
    { value: 'emergency', label: 'Emergency Leave - إجازة طارئة' },
    { value: 'unpaid', label: 'Unpaid Leave - إجازة بدون راتب' }
  ];

  // Get initial balance
  data.balance = getBalance(gs.getUserID(), 'annual');

  // Handle actions
  if (input && input.action) {

    if (input.action === 'getBalance') {
      data.balance = getBalance(gs.getUserID(), input.leave_type);
    }

    if (input.action === 'submitRequest') {
      var result = submitLeaveRequest(input.request);
      data.success = result.success;
      data.newRequest = result.record;
      data.error = result.error;
    }
  }

  function getBalance(employeeId, leaveType) {
    var util = new x_786686_hr_leave.LeaveBalanceUtil();
    return util.getBalance(employeeId, leaveType);
  }

  function submitLeaveRequest(req) {
    try {
      var gr = new GlideRecord('x_786686_hr_leave_request');
      gr.initialize();
      gr.employee = gs.getUserID();
      gr.leave_type = req.leave_type;
      gr.start_date = req.start_date;
      gr.end_date = req.end_date;
      gr.reason = req.reason || '';

      var sysId = gr.insert();

      if (sysId) {
        gr.get(sysId);
        return {
          success: true,
          record: {
            sys_id: sysId,
            number: gr.number.toString()
          }
        };
      }
      return { success: false, error: 'Insert failed' };
    } catch (e) {
      return { success: false, error: e.message };
    }
  }

})();

// Name: hrLeaveService
// Type: factory

function hrLeaveService($http, $q, spUtil) {

  var API_BASE = '/api/x_786686_hr_leave/hr_leave';

  return {

    getRequests: function(filters) {
      return $http.get(API_BASE + '/requests', { params: filters })
        .then(function(response) {
          return response.data.result;
        });
    },

    getRequest: function(sysId) {
      return $http.get(API_BASE + '/requests/' + sysId)
        .then(function(response) {
          return response.data.result;
        });
    },

    createRequest: function(request) {
      return $http.post(API_BASE + '/requests', request)
        .then(function(response) {
          return response.data.result;
        });
    },

    getBalance: function(leaveType) {
      return $http.get(API_BASE + '/balance', {
        params: { leave_type: leaveType }
      }).then(function(response) {
        return response.data.result.balance;
      });
    },

    getCalendar: function(month, year) {
      return $http.get(API_BASE + '/calendar', {
        params: { month: month, year: year }
      }).then(function(response) {
        return response.data.result;
      });
    }

  };

}

// Client Controller
api.controller = function($scope, hrLeaveService, spUtil) {
  var c = this;

  c.loading = true;

  hrLeaveService.getRequests().then(function(requests) {
    c.data.requests = requests;
    c.loading = false;
  }).catch(function(error) {
    spUtil.addErrorMessage('Failed to load requests');
    c.loading = false;
  });

  c.submitRequest = function() {
    hrLeaveService.createRequest(c.request).then(function(result) {
      spUtil.addInfoMessage('Request ' + result.number + ' created');
      c.data.requests.unshift(result);
    });
  };
};

  // Recent requests
  var gr = new GlideRecord('x_786686_hr_leave_request');
  gr.addQuery('employee', userId);
  gr.orderByDesc('sys_created_on');
  gr.setLimit(5);
  gr.query();

  data.recent = [];
  while (gr.next()) {
    data.recent.push({
      sys_id: gr.getUniqueValue(),
      number: gr.number.toString(),
      type: gr.leave_type.getDisplayValue(),
      dates: gr.start_date.getDisplayValue() + ' - ' + gr.end_date.getDisplayValue(),
      state: gr.state.getDisplayValue(),
      state_class: getStateClass(gr.state.toString())
    });
  }

  function getCount(query) {
    var ga = new GlideAggregate('x_786686_hr_leave_request');
    ga.addQuery('employee', userId);
    ga.addEncodedQuery(query);
    ga.addAggregate('COUNT');
    ga.query();
    return ga.next() ? parseInt(ga.getAggregate('COUNT')) : 0;
  }

  function getBalance(empId, type) {
    return new x_786686_hr_leave.LeaveBalanceUtil().getBalance(empId, type);
  }

  function getToday() {
    return new GlideDateTime().getDate().toString();
  }

  function getStateClass(state) {
    var classes = {'1': 'default', '2': 'warning', '3': 'success', '6': 'danger', '7': 'info'};
    return 'label-' + (classes[state] || 'default');
  }

})();

(function() {
  var userId = gs.getUserID();
  var util = new x_786686_hr_leave.LeaveBalanceUtil();
  var year = new GlideDateTime().getYearLocalTime();

  data.balances = [
    { 
      type: 'annual',
      label: 'Annual - سنوية',
      available: util.getBalance(userId, 'annual'),
      total: getAllocation(userId, 'annual'),
      percentage: 0
    },
    {
      type: 'sick',
      label: 'Sick - مرضية',
      available: util.getBalance(userId, 'sick'),
      total: getAllocation(userId, 'sick'),
      percentage: 0
    },
    {
      type: 'emergency',
      label: 'Emergency - طارئة',
      available: util.getBalance(userId, 'emergency'),
      total: getAllocation(userId, 'emergency'),
      percentage: 0
    }
  ];

  // Calculate percentages
  for (var i = 0; i < data.balances.length; i++) {
    var b = data.balances[i];
    b.percentage = b.total > 0 ? Math.round((b.available / b.total) * 100) : 0;
  }

  function getAllocation(empId, type) {
    var gr = new GlideRecord('x_786686_hr_leave_allocation');
    gr.addQuery('employee', empId);
    gr.addQuery('leave_type', type);
    gr.addQuery('year', year);
    gr.query();
    return gr.next() ? parseInt(gr.days_allocated) || 0 : 0;
  }
})();

// Server Script - Always validate user access
(function() {

  // Check user is authenticated
  if (!gs.isLoggedIn()) {
    data.error = 'Not authenticated';
    return;
  }

  // Only return user's own data (unless admin)
  var userId = gs.getUserID();
  var isAdmin = gs.hasRole('x_786686_hr_leave.hr_admin');

  if (input && input.employee_id) {
    if (input.employee_id !== userId && !isAdmin) {
      data.error = 'Access denied';
      return;
    }
  }

  // Validate input
  if (input && input.sys_id) {
    if (!/^[a-f0-9]{32}$/i.test(input.sys_id)) {
      data.error = 'Invalid ID format';
      return;
    }
  }

})();
