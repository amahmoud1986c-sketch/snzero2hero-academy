// Name: HR Leave - Validate Dates
// When: before insert, update

(function executeRule(current, previous) {

    var start = new GlideDateTime(current.start_date);
    var end = new GlideDateTime(current.end_date);
    var today = new GlideDateTime();

    if (start.before(today)) {
        current.setAbortAction(true);
        gs.addErrorMessage('Start date cannot be in the past');
        return;
    }

    if (end.before(start)) {
        current.setAbortAction(true);
        gs.addErrorMessage('End date must be after start date');
        return;
    }

    var days = GlideDateTime.subtract(start, end).getDayPart() + 1;
    if (days > 30) {
        current.setAbortAction(true);
        gs.addErrorMessage('Leave cannot exceed 30 days');
    }

})(current, previous);

var LeaveRequestUtil = Class.create();
LeaveRequestUtil.prototype = {

    initialize: function() {
        this.log = new GSLog('x_786686_hr_leave.log', 'LeaveRequestUtil');
    },

    createRequest: function(params) {
        var gr = new GlideRecord('x_786686_hr_leave_request');
        gr.initialize();
        gr.employee = params.employee;
        gr.leave_type = params.leaveType;
        gr.start_date = params.startDate;
        gr.end_date = params.endDate;
        gr.reason = params.reason || '';
        return gr.insert();
    },

    approveRequest: function(requestId, approverId) {
        var gr = new GlideRecord('x_786686_hr_leave_request');
        if (!gr.get(requestId)) return false;

        gr.state = 3;
        gr.approval_status = 'approved';
        gr.work_notes = 'Approved by ' + this._getUserName(approverId);
        gr.update();

        // Deduct balance
        new x_786686_hr_leave.LeaveBalanceUtil().deductBalance(
            gr.employee.toString(),
            gr.leave_type.toString(),
            parseFloat(gr.duration_days)
        );

        return true;
    },

    getEmployeeRequests: function(employeeId, filters) {
        var results = [];
        filters = filters || {};

        var gr = new GlideRecord('x_786686_hr_leave_request');
        gr.addQuery('employee', employeeId);
        if (filters.state) gr.addQuery('state', filters.state);
        gr.orderByDesc('sys_created_on');
        gr.query();

        while (gr.next()) {
            results.push({
                sys_id: gr.getUniqueValue(),
                number: gr.number.toString(),
                leave_type: gr.leave_type.getDisplayValue(),
                start_date: gr.start_date.getDisplayValue(),
                duration: parseFloat(gr.duration_days) || 0,
                state: gr.state.getDisplayValue()
            });
        }
        return results;
    },

    _getUserName: function(userId) {
        var user = new GlideRecord('sys_user');
        return user.get(userId) ? user.name.toString() : 'Unknown';
    },

    type: 'LeaveRequestUtil'
};

Leave Balance Utility

Open component:ApplicationLeave Requests
// Name: LeaveBalanceUtil
// API Name: x_786686_hr_leave.LeaveBalanceUtil

var LeaveBalanceUtil = Class.create();
LeaveBalanceUtil.prototype = {

    initialize: function() {},

    getBalance: function(employeeId, leaveType) {
        leaveType = leaveType || 'annual';
        var year = new GlideDateTime().getYearLocalTime();

        var alloc = new GlideRecord('x_786686_hr_leave_allocation');
        alloc.addQuery('employee', employeeId);
        alloc.addQuery('leave_type', leaveType);
        alloc.addQuery('year', year);
        alloc.query();

        if (alloc.next()) {
            var allocated = parseInt(alloc.days_allocated) || 0;
            var used = parseInt(alloc.days_used) || 0;
            return allocated - used;
        }
        return 0;
    },

    getAllBalances: function(employeeId) {
        var balances = {};
        var types = ['annual', 'sick', 'emergency', 'unpaid'];

        for (var i = 0; i < types.length; i++) {
            balances[types[i]] = this.getBalance(employeeId, types[i]);
        }
        return balances;
    },

    deductBalance: function(employeeId, leaveType, days) {
        var year = new GlideDateTime().getYearLocalTime();

        var alloc = new GlideRecord('x_786686_hr_leave_allocation');
        alloc.addQuery('employee', employeeId);
        alloc.addQuery('leave_type', leaveType);
        alloc.addQuery('year', year);
        alloc.query();

        if (alloc.next()) {
            alloc.days_used = (parseInt(alloc.days_used) || 0) + days;
            alloc.update();
            return true;
        }
        return false;
    },

    type: 'LeaveBalanceUtil'
};

OnChange Script

Open component:IncidentsProblemsChangesSOW
// Name: Calculate Duration
// Type: onChange
// Field: start_date, end_date

function onChange(control, oldValue, newValue, isLoading) {
    if (isLoading) return;

    var start = g_form.getValue('start_date');
    var end = g_form.getValue('end_date');

    if (!start || !end) return;

    var startDate = new Date(start);
    var endDate = new Date(end);

    if (endDate < startDate) {
        g_form.showFieldMsg('end_date', 'End must be after start', 'error');
        return;
    }

    var days = Math.ceil((endDate - startDate) / (1000 * 60 * 60 * 24)) + 1;
    g_form.setValue('duration_days', days);
    g_form.hideFieldMsg('end_date');

    checkBalance(days);
}

function checkBalance(days) {
    var ga = new GlideAjax('x_786686_hr_leave.LeaveAjaxProcessor');
    ga.addParam('sysparm_name', 'getBalance');
    ga.addParam('sysparm_employee_id', g_form.getValue('employee'));
    ga.addParam('sysparm_leave_type', g_form.getValue('leave_type'));

    ga.getXMLAnswer(function(answer) {
        var result = JSON.parse(answer);
        if (days > result.balance) {
            g_form.showFieldMsg('duration_days', 
                'Exceeds balance (' + result.balance + ')', 'error');
        }
    });
}

// Name: LeaveAjaxProcessor

var LeaveAjaxProcessor = Class.create();
LeaveAjaxProcessor.prototype = Object.extendsObject(AbstractAjaxProcessor, {

    getBalance: function() {
        var employeeId = this.getParameter('sysparm_employee_id');
        var leaveType = this.getParameter('sysparm_leave_type') || 'annual';

        var util = new x_786686_hr_leave.LeaveBalanceUtil();
        var balance = util.getBalance(employeeId, leaveType);

        return JSON.stringify({
            success: true,
            balance: balance
        });
    },

    getEmployeeData: function() {
        var employeeId = this.getParameter('sysparm_employee_id');

        var emp = new GlideRecord('sys_user');
        if (!emp.get(employeeId)) {
            return JSON.stringify({ error: 'Employee not found' });
        }

        var util = new x_786686_hr_leave.LeaveBalanceUtil();

        return JSON.stringify({
            success: true,
            name: emp.name.toString(),
            email: emp.email.toString(),
            manager_id: emp.manager.toString(),
            manager_name: emp.manager.getDisplayValue(),
            balance: util.getBalance(employeeId, 'annual')
        });
    },

    checkAvailability: function() {
        var employeeId = this.getParameter('sysparm_employee_id');
        var startDate = this.getParameter('sysparm_start_date');
        var endDate = this.getParameter('sysparm_end_date');

        var gr = new GlideRecord('x_786686_hr_leave_request');
        gr.addQuery('employee', employeeId);
        gr.addQuery('state', 'NOT IN', '6,7');
        var qc = gr.addQuery('start_date', '<=', endDate);
        qc.addCondition('end_date', '>=', startDate);
        gr.query();

        var conflicts = [];
        while (gr.next()) {
            conflicts.push({
                number: gr.number.toString(),
                dates: gr.start_date.getDisplayValue() + ' - ' + 
                       gr.end_date.getDisplayValue()
            });
        }

        return JSON.stringify({
            available: conflicts.length === 0,
            conflicts: conflicts
        });
    },

    type: 'LeaveAjaxProcessor'
});

(function process(request, response) {

    var employeeId = request.queryParams.employee_id;
    var state = request.queryParams.state;

    var gr = new GlideRecord('x_786686_hr_leave_request');
    if (employeeId) gr.addQuery('employee', employeeId);
    if (state) gr.addQuery('state', state);
    gr.orderByDesc('sys_created_on');
    gr.setLimit(100);
    gr.query();

    var results = [];
    while (gr.next()) {
        results.push({
            sys_id: gr.getUniqueValue(),
            number: gr.number.toString(),
            employee: gr.employee.getDisplayValue(),
            leave_type: gr.leave_type.getDisplayValue(),
            start_date: gr.start_date.getValue(),
            end_date: gr.end_date.getValue(),
            duration: parseFloat(gr.duration_days),
            state: gr.state.getDisplayValue()
        });
    }

    return { count: results.length, requests: results };

})(request, response);

// === POST /requests ===
(function process(request, response) {

    var body = request.body.data;

    // Validate
    if (!body.employee_id || !body.leave_type || 
        !body.start_date || !body.end_date) {
        response.setStatus(400);
        return { error: 'Missing required fields' };
    }

    var gr = new GlideRecord('x_786686_hr_leave_request');
    gr.initialize();
    gr.employee = body.employee_id;
    gr.leave_type = body.leave_type;
    gr.start_date = body.start_date;
    gr.end_date = body.end_date;
    gr.reason = body.reason || '';

    var sys_id = gr.insert();

    if (sys_id) {
        response.setStatus(201);
        gr.get(sys_id);
        return {
            success: true,
            sys_id: sys_id,
            number: gr.number.toString()
        };
    }

    response.setStatus(500);
    return { error: 'Failed to create' };

})(request, response);

var ExternalHRIntegration = Class.create();
ExternalHRIntegration.prototype = {

    initialize: function() {
        this.endpoint = gs.getProperty('x_786686_hr_leave.external_endpoint');
        this.apiKey = gs.getProperty('x_786686_hr_leave.api_key');
    },

    syncRequest: function(requestSysId) {
        var gr = new GlideRecord('x_786686_hr_leave_request');
        if (!gr.get(requestSysId)) return null;

        try {
            var rest = new sn_ws.RESTMessageV2();
            rest.setEndpoint(this.endpoint + '/leave-requests');
            rest.setHttpMethod('POST');
            rest.setRequestHeader('Authorization', 'Bearer ' + this.apiKey);
            rest.setRequestHeader('Content-Type', 'application/json');

            rest.setRequestBody(JSON.stringify({
                external_id: gr.number.toString(),
                employee: gr.employee.employee_number.toString(),
                type: gr.leave_type.toString(),
                start: gr.start_date.getValue(),
                end: gr.end_date.getValue(),
                days: parseFloat(gr.duration_days)
            }));

            var resp = rest.execute();
            if (resp.getStatusCode() == 201) {
                return JSON.parse(resp.getBody());
            }
        } catch (e) {
            gs.error('Sync error: ' + e.message);
        }
        return null;
    },

    type: 'ExternalHRIntegration'
};

Trigger: Record Updated
Table: x_786686_hr_leave_request
Condition: State changes to Submitted

Steps:
1. Look Up Record
   - Table: sys_user
   - ID: Trigger → Employee

2. If duration > 5 days
   - Ask Approval: Manager (2 days)
   - If Approved: Ask Approval: HR Admin

3. If All Approved:
   - Update Record: State = Approved
   - Send Email: Employee
   - Run Script: Deduct balance

4. If Rejected:
   - Update Record: State = Rejected
   - Send Email: Employee with reason
