Add User to Group

Open component:ACLsRolesGroupsUsers
function addUserToGroup(username, groupName) {
    var user = new GlideRecord('sys_user');
    var group = new GlideRecord('sys_user_group');

    if (!user.get('user_name', username) || !group.get('name', groupName)) {
        return false;
    }

    var member = new GlideRecord('sys_user_grmember');
    member.addQuery('user', user.sys_id);
    member.addQuery('group', group.sys_id);
    member.query();

    if (!member.hasNext()) {
        member.initialize();
        member.user = user.sys_id;
        member.group = group.sys_id;
        member.insert();
        return true;
    }
    return false;
}

Role Audit

Open component:ACLsRolesGroupsUsersRisksControlsIssuesAudit Engagements
function auditRoles(rolePrefix) {
    var report = [];

    var roles = new GlideRecord('sys_user_role');
    roles.addQuery('name', 'STARTSWITH', rolePrefix);
    roles.query();

    while (roles.next()) {
        var info = {
            name: roles.name.toString(),
            direct_users: 0,
            groups: []
        };

        // Count direct assignments
        var ur = new GlideAggregate('sys_user_has_role');
        ur.addQuery('role', roles.sys_id);
        ur.addAggregate('COUNT');
        ur.query();
        if (ur.next()) info.direct_users = parseInt(ur.getAggregate('COUNT'));

        // Get groups
        var gr = new GlideRecord('sys_group_has_role');
        gr.addQuery('role', roles.sys_id);
        gr.query();
        while (gr.next()) {
            info.groups.push(gr.group.getDisplayValue());
        }

        report.push(info);
    }
    return report;
}

function instanceHealthCheck() {
    var report = {
        timestamp: new GlideDateTime().getDisplayValue(),
        stats: {},
        warnings: []
    };

    // Active users
    var users = new GlideAggregate('sys_user');
    users.addQuery('active', true);
    users.addAggregate('COUNT');
    users.query();
    if (users.next()) {
        report.stats.active_users = parseInt(users.getAggregate('COUNT'));
    }

    // Open incidents
    var inc = new GlideAggregate('incident');
    inc.addQuery('state', 'NOT IN', '6,7');
    inc.addAggregate('COUNT');
    inc.query();
    if (inc.next()) {
        report.stats.open_incidents = parseInt(inc.getAggregate('COUNT'));
    }

    // Pending leave requests
    var leave = new GlideAggregate('x_786686_hr_leave_request');
    leave.addQuery('state', 2);
    leave.addAggregate('COUNT');
    leave.query();
    if (leave.next()) {
        var pending = parseInt(leave.getAggregate('COUNT'));
        report.stats.pending_leave = pending;
        if (pending > 50) {
            report.warnings.push('High pending leave requests: ' + pending);
        }
    }

    return report;
}

function getAuditTrail(table, recordId, days) {
    var trail = [];
    var since = new GlideDateTime();
    since.addDays(-days);

    var audit = new GlideRecord('sys_audit');
    audit.addQuery('tablename', table);
    audit.addQuery('documentkey', recordId);
    audit.addQuery('sys_created_on', '>=', since);
    audit.orderByDesc('sys_created_on');
    audit.query();

    while (audit.next()) {
        trail.push({
            field: audit.fieldname.toString(),
            old_value: audit.oldvalue.toString(),
            new_value: audit.newvalue.toString(),
            user: audit.user.getDisplayValue(),
            time: audit.sys_created_on.getDisplayValue()
        });
    }
    return trail;
}
