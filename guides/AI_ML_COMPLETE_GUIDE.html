Custom Server Script

// Virtual Agent Server Script
(function execute(inputs, outputs) {

    var userId = inputs.user_id;
    var leaveType = inputs.leave_type;
    var startDate = inputs.start_date;
    var endDate = inputs.end_date;
    var reason = inputs.reason;

    // Validate balance
    var util = new x_786686_hr_leave.LeaveBalanceUtil();
    var balance = util.getBalance(userId, leaveType);

    var start = new GlideDateTime(startDate);
    var end = new GlideDateTime(endDate);
    var duration = GlideDateTime.subtract(start, end).getDayPart() + 1;

    if (duration > balance) {
        outputs.success = false;
        outputs.message = 'Insufficient balance. Available: ' + balance + ' days';
        return;
    }

    // Create request
    var gr = new GlideRecord('x_786686_hr_leave_request');
    gr.initialize();
    gr.employee = userId;
    gr.leave_type = leaveType;
    gr.start_date = startDate;
    gr.end_date = endDate;
    gr.reason = reason;
    gr.duration_days = duration;

    var sysId = gr.insert();

    if (sysId) {
        gr.get(sysId);
        outputs.success = true;
        outputs.request_number = gr.number.toString();
        outputs.message = 'Request ' + gr.number + ' submitted successfully';
    } else {
        outputs.success = false;
        outputs.message = 'Failed to create request';
    }

})(inputs, outputs);

// VA Server Script with full AI integration
(function execute(inputs, outputs) {

    var assistant = new x_786686_hr_leave.LeaveAIAssistant();
    var userId = inputs.user_id;

    // Process natural language input
    var nluResult = assistant.processUserInput(inputs.user_message);

    outputs.intent = nluResult.intent;

    switch (nluResult.intent) {
        case 'request_leave':
            // Extract entities and create request
            var entities = nluResult.entities;
            outputs.response = 'I understand you want to request ' + 
                (entities.leave_type || 'leave') + '. ';
            outputs.next_action = 'collect_details';
            break;

        case 'check_balance':
            var util = new x_786686_hr_leave.LeaveBalanceUtil();
            var balances = util.getAllBalances(userId);
            outputs.response = 'Your leave balances:\n' +
                'Annual: ' + balances.annual + ' days\n' +
                'Sick: ' + balances.sick + ' days\n' +
                'Emergency: ' + balances.emergency + ' days';
            outputs.next_action = 'end';
            break;

        case 'view_requests':
            outputs.response = 'Here are your recent leave requests:';
            outputs.next_action = 'show_list';
            break;

        default:
            outputs.response = 'I\'m not sure what you need. Would you like to:\n' +
                '1. Request leave\n' +
                '2. Check your balance\n' +
                '3. View your requests';
            outputs.next_action = 'clarify';
    }

    // Add smart suggestions
    var suggestions = assistant.getSmartSuggestions(userId);
    if (suggestions.length > 0) {
        outputs.suggestions = suggestions;
    }

})(inputs, outputs);
