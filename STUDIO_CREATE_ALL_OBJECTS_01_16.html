/**
 * Script Include: HRLXProLeaveUtilsV2
 * Scope: x_786686_hr_leave
 * Application: Custom · Scope: x_786686_hr_leave · sys_id: 83d5298c932b8f107520703efaba1072
 * Client callable: true  (extends AbstractAjaxProcessor)
 * Purpose: Leave calculations + helpers (business days, approver, balance, audit).
 *
 * Create in ServiceNow:
 *   Studio > Create Application File > Script Include
 *   Name: HRLXProLeaveUtilsV2
 *   Client callable: checked
 */
var HRLXProLeaveUtilsV2 = Class.create();
HRLXProLeaveUtilsV2.prototype = Object.extendsObject(global.AbstractAjaxProcessor, {

    /**
     * Count business days (Mon-Fri) inclusive between two date strings.
     * @param {string} startDate - 'yyyy-MM-dd'
     * @param {string} endDate   - 'yyyy-MM-dd'
     * @return {number}
     */
    calculateBusinessDays: function (startDate, endDate) {
        if (!startDate || !endDate) return 0;
        var start = new GlideDateTime(startDate + ' 00:00:00');
        var end = new GlideDateTime(endDate + ' 00:00:00');
        if (end.before(start)) return 0;

        var days = 0;
        var cursor = new GlideDateTime(start);
        while (!cursor.after(end)) {
            var dow = cursor.getDayOfWeekLocalTime(); // 1=Mon ... 7=Sun
            if (dow >= 1 && dow <= 5) days++;
            cursor.addDaysLocalTime(1);
        }
        return days;
    },

    /**
     * Resolve the approver (employee's manager) and set it on the record.
     * @param {GlideRecord} currentRecord - leave request record
     */
    setApproverFromEmployee: function (currentRecord) {
        var empId = currentRecord.getValue('u_employee_ref');
        if (!empId) return;
        var emp = new GlideRecord('x_786686_hr_leave_employee_v2');
        if (emp.get(empId) && emp.u_manager) {
            currentRecord.u_approver_user = emp.getValue('u_manager');
        }
    },

    /**
     * Write an audit row when the state changes.
     */
    auditStateChange: function (leaveSysId, oldState, newState, eventType) {
        var a = new GlideRecord('x_786686_hr_leave_audit_v2');
        a.initialize();
        a.u_leave_ref = leaveSysId;
        a.u_old_state = oldState || '';
        a.u_new_state = newState || '';
        a.u_event_type = eventType || 'state_change';
        a.u_actor = gs.getUserID();
        a.insert();
    },

    /**
     * Recalculate remaining balance for an employee/year.
     */
    recalculateBalance: function (employeeSysId, year) {
        var used = this._sumApprovedDays(employeeSysId, year);
        var bal = new GlideRecord('x_786686_hr_leave_bal_v2');
        bal.addQuery('u_employee_ref', employeeSysId);
        bal.addQuery('u_year', year);
        bal.query();
        if (bal.next()) {
            bal.u_used = used;
            bal.u_remaining = parseInt(bal.getValue('u_entitled') || 0, 10) - used;
            bal.update();
        }
    },

    _sumApprovedDays: function (employeeSysId, year) {
        var total = 0;
        var gr = new GlideAggregate('x_786686_hr_leave_req_v2');
        gr.addQuery('u_employee_ref', employeeSysId);
        gr.addQuery('u_state', 'approved');
        gr.addQuery('u_start_date', '>=', year + '-01-01');
        gr.addQuery('u_start_date', '<=', year + '-12-31');
        gr.addAggregate('SUM', 'u_days_count');
        gr.query();
        if (gr.next()) total = parseInt(gr.getAggregate('SUM', 'u_days_count') || 0, 10);
        return total;
    },

    /**
     * GlideAjax entry point for client-side day calculation.
     */
    ajaxGetBusinessDays: function () {
        return this.calculateBusinessDays(
            this.getParameter('sysparm_start'),
            this.getParameter('sysparm_end')
        );
    },

    type: 'HRLXProLeaveUtilsV2'
});

/**
 * Script Include: HRLXProSecurityUtilsV2
 * Scope: x_786686_hr_leave
 * Application: Custom · Scope: x_786686_hr_leave · sys_id: 83d5298c932b8f107520703efaba1072
 * Client callable: false
 * Purpose: Centralized security checks used by ACLs.
 *
 * Used by ACL scripts, e.g.:
 *   answer = new x_786686_hr_leave.HRLXProSecurityUtilsV2().canReadLeave(current);
 */
var HRLXProSecurityUtilsV2 = Class.create();
HRLXProSecurityUtilsV2.prototype = {

    initialize: function () {
        this.userId = gs.getUserID();
    },

    /** HR admin can do anything. */
    isHrAdmin: function () {
        return gs.hasRole('x_786686_hr_leave.hr_admin');
    },

    /** Can the current user READ this leave request? */
    canReadLeave: function (current) {
        if (this.isHrAdmin()) return true;
        // owner
        if (current.getValue('u_requested_for_user') === this.userId) return true;
        // approver / manager
        if (current.getValue('u_approver_user') === this.userId) return true;
        // manager of the employee
        return this._isManagerOfEmployee(current.getValue('u_employee_ref'));
    },

    /** Can the current user WRITE this leave request? */
    canWriteLeave: function (current) {
        if (this.isHrAdmin()) return true;
        var state = current.getValue('u_state');
        // owner can edit only while draft/submitted
        if (current.getValue('u_requested_for_user') === this.userId) {
            return (state === 'draft' || state === 'submitted');
        }
        // approver can act when submitted
        if (current.getValue('u_approver_user') === this.userId) {
            return (state === 'submitted');
        }
        return false;
    },

    _isManagerOfEmployee: function (employeeSysId) {
        if (!employeeSysId) return false;
        var emp = new GlideRecord('x_786686_hr_leave_employee_v2');
        if (emp.get(employeeSysId)) {
            return emp.getValue('u_manager') === this.userId;
        }
        return false;
    },

    type: 'HRLXProSecurityUtilsV2'
};

/**
 * Application: Custom · Scope: x_786686_hr_leave · sys_id: 83d5298c932b8f107520703efaba1072
 * Scheduled Job: Annual Leave Accrual
 * Run: yearly (e.g. Jan 1)
 *
 * Creates a fresh balance row per employee for the new year if missing.
 * Default entitlement is read from a property (fallback 21 annual days).
 */
(function annualAccrual() {

    var year = new GlideDateTime().getYearLocalTime();
    var annualEntitlement = parseInt(gs.getProperty('x_786686_hr_leave.config.annual_days', '21'), 10);

    var emp = new GlideRecord('x_786686_hr_leave_employee_v2');
    emp.query();

    var created = 0;
    while (emp.next()) {
        // skip if a balance for this year/type already exists
        var existing = new GlideRecord('x_786686_hr_leave_bal_v2');
        existing.addQuery('u_employee_ref', emp.getUniqueValue());
        existing.addQuery('u_year', year);
        existing.addQuery('u_leave_type', 'annual');
        existing.query();
        if (existing.hasNext()) continue;

        var bal = new GlideRecord('x_786686_hr_leave_bal_v2');
        bal.initialize();
        bal.u_employee_ref = emp.getUniqueValue();
        bal.u_year = year;
        bal.u_leave_type = 'annual';
        bal.u_entitled = annualEntitlement;
        bal.u_used = 0;
        bal.u_remaining = annualEntitlement;
        bal.insert();
        created++;
    }

    gs.info('[HRLX] Annual accrual ' + year + ' created ' + created + ' balance rows.');
})();

/**
 * Script Include: HRLXProIntegrationUtilsV2
 * Scope: x_786686_hr_leave
 * Application: Custom · Scope: x_786686_hr_leave · sys_id: 83d5298c932b8f107520703efaba1072
 * Client callable: false
 * Purpose: Inbound create + Outbound push for leave integration.
 */
var HRLXProIntegrationUtilsV2 = Class.create();
HRLXProIntegrationUtilsV2.prototype = {

    initialize: function () {},

    /**
     * Inbound: create a leave request from a JSON payload (string).
     * @return {string} sys_id of created record
     */
    inboundCreateLeave: function (jsonString) {
        var data = {};
        try { data = JSON.parse(jsonString); } catch (e) { return ''; }

        var gr = new GlideRecord('x_786686_hr_leave_req_v2');
        gr.initialize();
        gr.u_employee_ref = this._findEmployeeByExternalId(data.employee_id);
        gr.u_requested_for_user = this._findUserByEmployee(data.employee_id);
        gr.u_leave_type = data.leave_type || 'annual';
        gr.u_start_date = data.start_date || '';
        gr.u_end_date = data.end_date || '';
        gr.u_reason = data.reason || '';
        gr.u_state = 'submitted';
        return gr.insert();
    },

    /**
     * Outbound: push an approved leave decision to ERP/Payroll.
     * Endpoint + auth come from system properties (no secrets in code).
     */
    outboundPushDecision: function (leaveSysId) {
        var gr = new GlideRecord('x_786686_hr_leave_req_v2');
        if (!gr.get(leaveSysId)) return false;

        var endpoint = gs.getProperty('x_786686_hr_leave.config.erp_endpoint');
        if (!endpoint) {
            gs.error('[HRLX] Missing property x_786686_hr_leave.config.erp_endpoint');
            return false;
        }

        try {
            var r = new sn_ws.RESTMessageV2();
            r.setEndpoint(endpoint);
            r.setHttpMethod('POST');
            r.setRequestHeader('Content-Type', 'application/json');
            r.setRequestBody(JSON.stringify({
                number: gr.getValue('u_number'),
                employee: gr.getDisplayValue('u_employee_ref'),
                type: gr.getValue('u_leave_type'),
                days: gr.getValue('u_days_count'),
                state: gr.getValue('u_state')
            }));
            var resp = r.execute();
            var code = resp.getStatusCode();
            gs.info('[HRLX] Outbound status ' + code);
            return (code >= 200 && code < 300);
        } catch (ex) {
            gs.error('[HRLX] Outbound error: ' + ex.message);
            return false;
        }
    },

    _findEmployeeByExternalId: function (employeeId) {
        var e = new GlideRecord('x_786686_hr_leave_employee_v2');
        e.addQuery('u_employee_id', employeeId);
        e.query();
        return e.next() ? e.getUniqueValue() : '';
    },

    _findUserByEmployee: function (employeeId) {
        var e = new GlideRecord('x_786686_hr_leave_employee_v2');
        e.addQuery('u_employee_id', employeeId);
        e.query();
        return e.next() ? e.getValue('u_user') : '';
    },

    type: 'HRLXProIntegrationUtilsV2'
};

(function process(request, response) {

    // Validate

    response.setStatus(201);
    response.setBody({ sys_id: sysId, message: 'Leave request created' });
