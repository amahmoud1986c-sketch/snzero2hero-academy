3.1 Script Include — Track A (leave, read-only)

var LabAIReadTools = Class.create();
LabAIReadTools.prototype = {
    initialize: function() {
        this.T_REQ = 'x_786686_hr_leave_req_v2';
        this.T_EMP = 'x_786686_hr_leave_employee_v2';
    },

    _employeeSysIdForUser: function(userId) {
        var gr = new GlideRecordSecure(this.T_EMP);
        gr.addQuery('user', userId);
        gr.setLimit(1);
        gr.query();
        return gr.next() ? gr.getUniqueValue() : null;
    },

    /** AI Agent tool — list leave requests visible to invoking user via ACLs */
    listLeaveForUser: function() {
        var userId = gs.getUserID();
        var empId = this._employeeSysIdForUser(userId);
        if (!empId) return { ok: true, rows: [], note: 'no_employee_profile' };

        var out = [];
        var req = new GlideRecordSecure(this.T_REQ);
        req.addQuery('u_employee_ref', empId);
        req.orderByDesc('sys_created_on');
        req.setLimit(25);
        req.query();
        while (req.next()) {
            out.push({
                number: req.getValue('number'),
                state: req.getDisplayValue('u_state'),
                type: req.getDisplayValue('u_leave_type'),
                start: req.getValue('u_start_date'),
                end: req.getValue('u_end_date'),
                days: req.getValue('u_days')
            });
        }
        return { ok: true, count: out.length, rows: out };
    },

    type: 'LabAIReadTools'
};

3.2 Script Include — Track B (incidents, read-only)

var LabAIReadTools = Class.create();
LabAIReadTools.prototype = {
    initialize: function() {},

    /** AI Agent tool — open incidents the invoking user may read (ACL-enforced) */
    listOpenIncidents: function(limit) {
        var max = Math.min(parseInt(limit, 10) || 10, 25);
        var out = [];
        var inc = new GlideRecordSecure('incident');
        inc.addActiveQuery();
        inc.orderByDesc('opened_at');
        inc.setLimit(max);
        inc.query();
        while (inc.next()) {
            out.push({
                number: inc.getValue('number'),
                short_description: inc.getValue('short_description'),
                state: inc.getDisplayValue('state'),
                priority: inc.getDisplayValue('priority'),
                assignment_group: inc.getDisplayValue('assignment_group')
            });
        }
        return { ok: true, count: out.length, rows: out };
    },

    getIncidentByNumber: function(number) {
        if (!number) return { ok: false, error: 'number_required' };
        var inc = new GlideRecordSecure('incident');
        inc.addQuery('number', number);
        inc.setLimit(1);
        inc.query();
        if (!inc.next()) return { ok: false, error: 'not_found_or_denied' };
        return {
            ok: true,
            number: inc.getValue('number'),
            description: inc.getValue('description'),
            state: inc.getDisplayValue('state'),
            opened_at: inc.getValue('opened_at')
        };
    },

    type: 'LabAIReadTools'
};

v1 rule: do not register requestLeave, state changes, or Flow actions yet. First release = read + summarize only. Add writes only after ACL review and Control Tower registration.

Pass criteria: test trace shows tool call → JSON rows → natural language answer; impersonating a user with no rows returns empty list, not an error stack; no field named u_state or incident.state is updated by the tool.

5. RAG — practical steps (high level)

Retrieval Augmented Generation = give the model relevant text at question time; do not rely on training memory. On ServiceNow you usually implement retrieval first, generation second.

IngestKB articles, policy docs, scoped tables — only published/current sources.
IndexAI Search application + sources + relevancy profile.
RetrieveTop-k chunks with minimum score; respect user criteria + ACLs.
GenerateNow Assist skill, AI Search gen answer, or agent tool (search_retrieval / custom Script Include).

6. MCP — practical pointer

8. Best practices — do / don’t

✓ GlideRecordSecure in tools ✓ Read-mostly first release ✓ Register in Control Tower ✓ Export test traces ✓ Narrow tool descriptions ✗ Tools write u_state / assignment ✗ Bypass ACLs in Script Include ✗ Shadow API keys in chat ✗ Unregistered agents in prod ✗ GlideRecord without Secure

First release contract: agents may read and explain; humans or approved Flows still commit state changes until security sign-off.

9. ServiceNow SDK — what it CAN and CANNOT do (AI agents)

From SDK 4.4+, Fluent APIs include AiAgent, agent tools, triggers, and agentic workflow artefacts — same governance tables as UI (sn_aia_*), expressed as TypeScript in your repo.

SDK CANSDK CANNOT
Define agents, tools, triggers as code; review in PRs; deploy with npx @servicenow/sdk build/deploy.
Follow official topics: building-ai-agents-guide, building-ai-agents-tools-guide, building-ai-agents-advanced-guide (see SDK Cursor documentation topic list). 		Replace AI Control Tower governance (inventory, policy, monitoring still required).
Invent arbitrary tools without ACL allow-list and scoped Script Includes — runtime still enforces platform security.
Skip licensing — SKUs must exist on the target instance.
