Step-by-step create

  1. ATF → Tests → New → Name: HRLX BR1 Days Calculation.
  2. Step 1: Create a Record → Table req_v2 → u_start_date=2026-06-01, u_end_date=2026-06-05, u_state=draft.
  3. Step 2: Run Server Side Script → assertion below.

Full script (assertion step)

(function(outputs, steps, stepResult, assertEqual) {
  var gr = new GlideRecord('x_786686_hr_leave_req_v2');
  gr.get(steps('Create a Record').record_id);
  assertEqual({ name: 'days', expected: '5', actual: gr.getValue('u_days_count') });
})(outputs, steps, stepResult, assertEqual);
